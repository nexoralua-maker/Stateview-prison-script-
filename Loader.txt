local function findMeshAndTexture(node)
    if not node or type(node) ~= "table" then return nil end
    local props = node.Props
    if props then
        local meshId = props.MeshId or props.MeshID or props.SearchId or props.SearchID
        if meshId and meshId ~= "" then
            return {
                meshid = meshId,
                textureid = props.TextureId or props.TextureID or "",
                scale = props.Scale or Vector3.new(1, 1, 1),
                size = props.Size or Vector3.new(0.2697, 1.2582, 4.2087),
                kind = props.Scale and "FileMesh" or "MeshPart"
            }
        end
    end
    if node.Display and type(node.Display) == "table" then
        local partSize
        for _, child in ipairs(node.Display) do
            local p = child.Props
            if p and p.Size then partSize = p.Size end
        end
        for _, child in ipairs(node.Display) do
            local p = child.Props
            if p then
                local mid = p.MeshId or p.MeshID or p.SearchId or p.SearchID
                if mid and mid ~= "" then
                    return {
                        meshid = mid,
                        textureid = p.TextureId or p.TextureID or "",
                        scale = p.Scale or Vector3.new(1, 1, 1),
                        size = p.Size or partSize or Vector3.new(0.2697, 1.2582, 4.2087),
                        kind = (child.Class == "MeshPart") and "MeshPart" or "FileMesh"
                    }
                end
            end
        end
    end
    return nil
end
local function normName(s)
    s = tostring(s or ""):lower()
    s = s:gsub("[%s%-%_]+", "")
    return s
end
local DataIndex = {}
local function buildDataIndex()
    DataIndex = {}
    for key, entry in pairs(data) do
        if type(entry) == "table" then
            DataIndex[normName(key)] = DataIndex[normName(key)] or key
            local base = tostring(key):gsub("_G_%d+$", ""):gsub("_K_%d+$", ""):gsub("_%d+$", "")
            DataIndex[normName(base)] = DataIndex[normName(base)] or key
            if entry.Meta and entry.Meta.ItemName then DataIndex[normName(entry.Meta.ItemName)] = DataIndex[normName(entry.Meta.ItemName)] or key end
        end
    end
end
buildDataIndex()
local ToolNameKnown = {}
local function getWeaponData(name)
    if not name then return nil end
    if data[name] then return findMeshAndTexture(data[name]) end
    local key = DataIndex[normName(name)]
    if key and data[key] then return findMeshAndTexture(data[key]) end
    return nil
end
local function resolveToolName(tn, itemType)
    if not tn or tn == "" then return nil end
    if getWeaponData(tn) then return tn end
    local base = tostring(tn):gsub("[%s%-%_]*[Gg][Uu][Nn]$", "")
    base = base:gsub("[%s%-%_]*[Kk][Nn][Ii][Ff][Ee]$", "")
    if base == "" or base == tn then return nil end
    local key = DataIndex[normName(base)]
    local en = key and data[key]
    local mt = en and en.Meta and en.Meta.ItemType
    if mt == itemType and getWeaponData(base) then return base end
    return nil
end

local function hideSpawnUIs()
    local kept = {}
    pcall(function()
        local me = game.Players.LocalPlayer
        local pg = me and me:FindFirstChild("PlayerGui")
        if not pg then return end
        for _, g in pairs(pg:GetChildren()) do
            if g:IsA("ScreenGui") then
                local n = string.lower(tostring(g.Name))
                if n ~= "gingerspawner" and n ~= "gingersplash" and n ~= "gingerwarm"
                    and (n:find("spawn") or n:find("fade") or n:find("death") or n:find("dead")
                        or n:find("respawn") or n:find("health") or n:find("damage")
                        or n:find("vignette") or n:find("hurt") or n == "maingui") then
                    local was = nil
                    pcall(function()
                        was = g.Enabled
                        g.Enabled = false
                    end)
                    kept[#kept + 1] = { g, was }
                end
            end
        end
    end)
    return kept
end
local function restoreSpawnUIs(kept, delaySecs)
    if type(kept) ~= "table" then return end
    task.delay(delaySecs or 2.5, function()
        if MY_RUN ~= getgenv().GingerRunId then return end
        pcall(function()
            for _, rec in pairs(kept) do
                local g, was = rec[1], rec[2]
                if g and g.Parent then
                    if was == nil then was = true end
                    g.Enabled = was
                end
            end
        end)
    end)
end
_G.GingerSilentRespawn = function()
    if MY_RUN ~= getgenv().GingerRunId then return false end
    local lastTool = _G.GingerLastRoleToolAt
    if lastTool and os.clock() - lastTool < 8 and _G.GingerEndVisible ~= true then return false, "mid-round" end
    local endNow = _G.GingerEndVisible == true
    local endFresh = (os.clock() - (_G.GingerEndSeenAt or 0)) < 60
    local latchLive = false
    pcall(function()
        local tbl = _G.GingerRelayRoles
        if type(tbl) == "table" then
            for _, rec in pairs(tbl) do
                if type(rec) == "table" and rec.role == "Murderer" then
                    latchLive = true
                    break
                end
            end
        end
        if not latchLive or _G.GingerRelayMurderer == nil then
            latchLive = false
        end
    end)
    if latchLive and not endNow then return false, "mid-round" end
    if _G.GingerRoundLive == true and not endNow and not endFresh then
        local sinceStart = os.clock() - (_G.GingerRoundStartAt or 0)
        local toolsQuiet = (os.clock() - (_G.GingerLastRoleToolAt or os.clock())) > 30
        if sinceStart <= 90 or not toolsQuiet then return false, "mid-round" end
    end
    if not endNow and not endFresh then
        if os.clock() - (_G.GingerTeleportAt or 0) < 35 then return false, "mid-round" end
        if os.clock() - (_G.GingerBootAt or 0) < 20 then return false, "warming up" end
    end
    if _G.GingerRespawning then return false, "busy" end
    local me = game.Players.LocalPlayer
    local ch, hum = nil, nil
    for i = 1, 10 do
        if MY_RUN ~= getgenv().GingerRunId then return false, "stale run" end
        ch = me and me.Character
        hum = ch and ch:FindFirstChildOfClass("Humanoid")
        if hum then break end
        task.wait(0.5)
    end
    if not hum then return false, "no character" end
    local cf = nil
    pcall(function()
        local root = ch:FindFirstChild("HumanoidRootPart")
        if root then cf = root.CFrame end
    end)
    _G.GingerRespawnRestore = cf
    _G.GingerRespawning = true
    _G.GingerIgnorePrint = true
    _G.GingerSilentActive = true
    _G.GingerHiddenUIs = hideSpawnUIs()
    pcall(function()
        local sg = game:GetService("StarterGui")
        _G.GingerCoreHealthWas = sg:GetCoreGuiEnabled(Enum.CoreGuiType.Health)
        sg:SetCoreGuiEnabled(Enum.CoreGuiType.Health, false)
    end)
    local mode = "death"
    pcall(function()
        if _G.GingerSilentMode == "pop" then mode = "pop" end
    end)
    local ok = false
    if mode == "pop" then
        pcall(function()
            me:LoadCharacter()
            ok = true
        end)
    else
        pcall(function()
            local conn = nil
            conn = hum.Died:Connect(function()
                pcall(function()
                    if conn then conn:Disconnect() end
                end)
                if MY_RUN ~= getgenv().GingerRunId then return end
                pcall(function()
                    local dc = me.Character
                    if dc then
                        for _, d in pairs(dc:GetDescendants()) do
                            if d:IsA("Sound") then
                                pcall(function() d.Volume = 0 end)
                                pcall(function() d:Destroy() end)
                            end
                        end
                    end
                end)
                pcall(function() me:LoadCharacter() end)
            end)
        end)
        pcall(function()
            hum:ChangeState(Enum.HumanoidStateType.Dead)
            ok = true
        end)
    end
    if not ok then
        _G.GingerRespawnRestore = nil
        _G.GingerRespawning = nil
        _G.GingerSilentActive = nil
        pcall(function()
            game:GetService("StarterGui"):SetCoreGuiEnabled(Enum.CoreGuiType.Health,
                _G.GingerCoreHealthWas ~= false)
        end)
        _G.GingerCoreHealthWas = nil
        if _G.GingerHiddenUIs then
            local h = _G.GingerHiddenUIs
            _G.GingerHiddenUIs = nil
            restoreSpawnUIs(h, 0.2)
        end
    else
        task.delay(8, function()
            if MY_RUN ~= getgenv().GingerRunId then return end
            _G.GingerRespawning = nil
        end)
    end
    if ok then return true end
    return false, "failed"
end
local GINGER_ID = "15374602183"
local GINGER_DISPLAY_SCALE = 0.09
local function applyWeaponMesh(refPart, weaponData)
    if not weaponData or not weaponData.meshid then return end
    local isGinger = tostring(weaponData.meshid):find(GINGER_ID, 1, true) ~= nil
    if refPart:IsA("MeshPart") then
        local specialMesh = refPart:FindFirstChildOfClass("SpecialMesh")
        if specialMesh then pcall(function() specialMesh:Destroy() end) end
        refPart.Transparency = 0
        pcall(function() refPart.Size = weaponData.size end)
        refPart.MeshId = weaponData.meshid
        refPart.TextureID = weaponData.textureid
    else
        local mesh = refPart:FindFirstChildOfClass("SpecialMesh")
        if not mesh then
            mesh = Instance.new("SpecialMesh")
            mesh.Name = "Mesh"
            mesh.MeshType = Enum.MeshType.FileMesh
            mesh.Parent = refPart
        elseif mesh.MeshType ~= Enum.MeshType.FileMesh then
            mesh.MeshType = Enum.MeshType.FileMesh
        end
        refPart.Transparency = 0
        pcall(function() refPart.Size = weaponData.size end)
        mesh.MeshId = weaponData.meshid
        mesh.TextureId = weaponData.textureid
        if isGinger then
            mesh.Scale = Vector3.new(GINGER_DISPLAY_SCALE, GINGER_DISPLAY_SCALE, GINGER_DISPLAY_SCALE)
        else
            mesh.Scale = weaponData.scale
        end
    end
end

local SyncNorm = nil
local function grantWeapons()
    local grantedKey = false
    pcall(function()
        if SyncNorm == nil then
            SyncNorm = {}
            for sname, it in pairs(Sync.Weapons) do
                SyncNorm[normName(sname)] = sname
                if type(it) == "table" then
                    if it.ItemName then
                        local k = normName(it.ItemName)
                        if not SyncNorm[k] then SyncNorm[k] = sname end
                    end
                    if it.Name then
                        local k = normName(it.Name)
                        if not SyncNorm[k] then SyncNorm[k] = sname end
                    end
                end
            end
        end
        local want = {}
        local function addPick(name)
            if type(name) ~= "string" or name == "" then return end
            local sk = SyncNorm[normName(name)]
            if sk == nil then
                local dk = DataIndex[normName(name)]
                if dk and Sync.Weapons[dk] then sk = dk end
            end
            if sk and Sync.Weapons[sk] then want[sk] = true end
        end
        addPick(_G.GingerPickGun)
        addPick(_G.GingerPickKnife)
        for k in pairs(_G.GingerSpawnedSet) do
            if Sync.Weapons[k] then want[k] = true end
        end
        for key in pairs(want) do
            local itemData = Sync.Weapons[key]
            itemData.Rarity = itemData.Rarity or "Common"
            if Sync.Rarities[itemData.Rarity] then
                local weaponMeshInfo = getWeaponData(key)
                if weaponMeshInfo and weaponMeshInfo.meshid and weaponMeshInfo.textureid
                    and weaponMeshInfo.scale and weaponMeshInfo.size then
                    ProfileData.Weapons.Owned[key] = 1
                    _G.GingerSpawnedSet[key] = true
                    grantedKey = key
                end
            end
        end
    end)
    return grantedKey
end

local function dressOwnStuckKnife(part)
    if not (part and part:IsA("BasePart") and part.Name == "StuckKnife") then return end
    local ours = (os.clock() - lastLocalThrowAt <= 6)
    if not ours then
        pcall(function()
            local me = game.Players.LocalPlayer
            if me and me:GetAttribute("GingerHeldKnife") == true then
                local has = false
                local ch = me.Character
                if ch then
                    for _, t in pairs(ch:GetChildren()) do
                        if t:IsA("Tool") and t:GetAttribute("ItemType") == "Knife" then
                            has = true
                            break
                        end
                    end
                end
                if not has then
                    local bp = me:FindFirstChildOfClass("Backpack")
                    if bp then
                        for _, t in pairs(bp:GetChildren()) do
                            if t:IsA("Tool") and t:GetAttribute("ItemType") == "Knife" then
                                has = true
                                break
                            end
                        end
                    end
                end
                ours = not has
            end
        end)
    end
    if not ours then
        pcall(function()
            local me = game.Players.LocalPlayer
            if me and me:GetAttribute("GingerWasMurderer") == true then
                ours = true
            end
        end)
    end
    if not ours then return end
    local equippedKnifeName = nil
    pcall(function()
        equippedKnifeName = ProfileData.Weapons.Equipped.Knife
    end)
    if not equippedKnifeName or equippedKnifeName == "" then return end
    local knifedata = getWeaponData(equippedKnifeName)
    if not (knifedata and knifedata.meshid) then return end
    local function apply()
        if not part.Parent then return end
        if part:IsA("MeshPart") then
            pcall(function()
                part.MeshId = knifedata.meshid
                part.TextureID = knifedata.textureid or ""
            end)
        end
        local mesh = part:FindFirstChildOfClass("SpecialMesh")
            or part:FindFirstChild("Mesh")
        if mesh and mesh:IsA("SpecialMesh") then
            pcall(function()
                mesh.MeshType = Enum.MeshType.FileMesh
                mesh.MeshId = knifedata.meshid
                mesh.TextureId = knifedata.textureid or ""
                if knifedata.scale then mesh.Scale = knifedata.scale end
            end)
        end
    end
    apply()
    task.spawn(function()
        for _ = 1, 20 do
            if not part.Parent then return end
            apply()
            task.wait(0.05)
        end
    end)
end

local function localHideOnAdd(tool)
    if not (tool and tool:IsA("Tool")) then return end
    if tool:GetAttribute("GingerHideHidden") ~= nil then return end
    local h = tool:FindFirstChild("Handle")
    if not (h and h:IsA("BasePart")) then return end
    tool:SetAttribute("GingerHideHidden", true)
    tool:SetAttribute("GingerHideBornAt", os.clock())
    pcall(function()
        if h.Transparency ~= 1 then
            tool:SetAttribute("GingerHideTransp", h.Transparency)
        end
        h.Transparency = 1
    end)
end
local function localRevealOnWeld(tool)
    if not (tool and tool:IsA("Tool")) then return end
    if tool:GetAttribute("GingerHideHidden") == nil then return end
    local ch = game.Players.LocalPlayer.Character
    local h = tool:FindFirstChild("Handle")
    if not ch or not (h and h:IsA("BasePart")) then return end
    local bornAt = tool:GetAttribute("GingerHideBornAt") or 0
    local welded = false
    if os.clock() - bornAt > 0.5 then
        welded = true
    else
        pcall(function()
            for _, w in pairs(ch:GetDescendants()) do
                if w:IsA("Weld") or w:IsA("Motor6D") or w:IsA("WeldConstraint") then
                    local p0, p1 = w.Part0, w.Part1
                    if (p0 == h and p1 and p1:IsDescendantOf(ch) and p1 ~= h)
                        or (p1 == h and p0 and p0:IsDescendantOf(ch) and p0 ~= h) then
                        welded = true
                        break
                    end
                end
            end
        end)
    end
    if welded then
        tool:SetAttribute("GingerHideHidden", nil)
        tool:SetAttribute("GingerHideBornAt", nil)
        local orig = tool:GetAttribute("GingerHideTransp")
        pcall(function()
            h.Transparency = (typeof(orig) == "number") and orig or 0
        end)
        tool:SetAttribute("GingerHideTransp", nil)
    else
        pcall(function() h.Transparency = 1 end)
    end
end
local function dressToolNow(tool, equippedGunName, equippedKnifeName, held)
    if not (tool and tool:IsA("Tool")) then return end
    local itemType = tool:GetAttribute("ItemType")
    local Handle = tool:FindFirstChild("Handle")
    if not Handle then return end
    if itemType == "Knife" or itemType == "Gun" then
        local tn = tool.Name
        if tn and tn ~= "" then
            local known = ToolNameKnown[tn]
            if known == nil then
                known = resolveToolName(tn, itemType) or false
                ToolNameKnown[tn] = known
            end
            if known then
                if itemType == "Knife" then equippedKnifeName = known
                else equippedGunName = known end
            end
        end
    end
    if itemType == "Gun" then
        local t = nil
        pcall(function() t = ProfileData.Weapons.Equipped.Gun end)
        if t and t ~= "" and getWeaponData(t) then equippedGunName = t end
    elseif itemType == "Knife" then
        local t = nil
        pcall(function() t = ProfileData.Weapons.Equipped.Knife end)
        if t and t ~= "" and getWeaponData(t) then equippedKnifeName = t end
    end
    if itemType == "Knife" and equippedKnifeName then
        local knifedata = getWeaponData(equippedKnifeName)
        if knifedata and knifedata.meshid and knifedata.textureid and knifedata.scale and knifedata.size then
            applyWeaponMesh(Handle, knifedata)
        end
        if held then localRevealOnWeld(tool) end
    elseif itemType == "Gun" and equippedGunName then
        local gundata = getWeaponData(equippedGunName)
        if gundata and gundata.meshid and gundata.textureid and gundata.scale and gundata.size then
            applyWeaponMesh(Handle, gundata)
        end
        if held then localRevealOnWeld(tool) end
    elseif held then
        localRevealOnWeld(tool)
    end
end

        end)
        pcall(function()
            tool.Equipped:Connect(function()

                localHideOnAdd(tool)
                local eqG, eqK = equippedNames()
                dressToolNow(tool, eqG, eqK, true)
            end)
        end)
    end

        if not c then return end
        for _, t in pairs(c:GetChildren()) do hookToolDress(t) end
        if hookedDress[c] then return end

        pcall(function()
            c.ChildAdded:Connect(function(added)
                if MY_RUN ~= getgenv().GingerRunId then return end

                    local eqG, eqK = equippedNames()

                    dressToolNow(added, eqG, eqK, added.Parent == chNow)
                end
            end)
        end)

local function peerToolsOut()
    local out = false
    pcall(function()
        for _, plr in pairs(game.Players:GetPlayers()) do
            local c = plr.Character
            if c then
                for _, t in pairs(c:GetChildren()) do
                    if t:IsA("Tool") then
                        local it = t:GetAttribute("ItemType")
                        if it == "Knife" or it == "Gun" then
                            out = true
                            break
                        end
                    end
                end
            end
            if out then break end
        end
    end)
    return out
end
local function unmaskPeer(pname)
    local rec = _G.GingerDecoys and _G.GingerDecoys[pname] or nil
    if _G.GingerDecoys then _G.GingerDecoys[pname] = nil end
    if type(rec) ~= "table" then return end
    pcall(function()
        if rec.model then rec.model:Destroy() end
    end)
    pcall(function()
        if type(rec.hidden) == "table" then
            for _, h in pairs(rec.hidden) do
                if h[1] and h[1].Parent then
                    if h[2] == "t" or h[2] == "dt" then h[1].Transparency = h[3] end
                end
            end
        end
    end)
end
local function maskPeerDeath(plr, ch)
    if MY_RUN ~= getgenv().GingerRunId then return end
    if not plr or plr == game.Players.LocalPlayer then return end
    local pname = plr.Name
    if _G.GingerDecoys[pname] then return end
    if peerToolsOut() then return end
    local dc = nil
    pcall(function()
        ch.Archivable = true
        dc = ch:Clone()
    end)
    if not dc then return end
    pcall(function()
        for _, d in pairs(dc:GetDescendants()) do
            if d:IsA("Humanoid") or d:IsA("Animator") then
                d:Destroy()
            elseif d:IsA("LocalScript") or d:IsA("Script") then
                d:Destroy()
            elseif d:IsA("BasePart") then
                d.Anchored = true
                d.CanCollide = false
                d.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            end
        end
        dc.Name = "GingerDecoy_" .. pname
        dc.Parent = game.Workspace
    end)
    if not dc.Parent then
        pcall(function() dc:Destroy() end)
        return
    end
    local hidden = {}
    pcall(function()
        for _, d in pairs(ch:GetDescendants()) do
            if d:IsA("BasePart") then
                hidden[#hidden + 1] = { d, "t", d.Transparency }
                d.Transparency = 1
            elseif d:IsA("Decal") or d:IsA("Texture") then
                hidden[#hidden + 1] = { d, "dt", d.Transparency }
                d.Transparency = 1
            end
        end
    end)
    local spot = nil
    pcall(function()
        local r = ch:FindFirstChild("HumanoidRootPart")
        if r then spot = r.CFrame end
    end)
    _G.GingerDecoys[pname] = { model = dc, hidden = hidden, at = os.clock(), spot = spot }
    task.spawn(function()
        local t0 = os.clock()
        while MY_RUN == getgenv().GingerRunId do
            local done = false
            pcall(function()
                if not (_G.GingerDecoys and _G.GingerDecoys[pname]) then
                    done = true
                    return
                end
                if os.clock() - t0 > 6 then
                    done = true
                    return
                end
                local nc = plr.Character
                local nr = nc and nc:FindFirstChild("HumanoidRootPart")
                local sp = (_G.GingerDecoys[pname] or {}).spot
                if nr and sp and (nr.Position - sp.Position).Magnitude < 20 then
                    done = true
                    return
                end
            end)
            if done then break end
            task.wait(0.2)
        end
        if MY_RUN == getgenv().GingerRunId then
            unmaskPeer(pname)
        end
    end)
end

                        local ok = true
                        local p = d
                        while p and p ~= pg do
                            if p:IsA("GuiObject") and not p.Visible then
                                ok = false
                                break
                            end
                            if p:IsA("ScreenGui") and not p.Enabled then
                                ok = false
                                break

                        end

                        end
                    end

            end
            _G.GingerEndVisible = vis

                or (os.clock() - (_G.GingerRestoreAt or 0) < 3)
