-- ============================================================
-- ARMOR MODULE
-- ============================================================
local ArmorModule = {}
local ARMOR_ATTRIBUTES      = {"Armor", "ArmorPercent", "ArmorPercentage", "ArmorValue", "Shield"}
local ARMOR_MAX_ATTRIBUTES  = {"MaxArmor", "ArmorMax", "MaxShield"}
local ARMOR_VALUES          = {"Armor", "ArmorValue", "Shield"}
local DEFAULT_MAX_ARMOR     = 100

local function findAttribute(container, names)
    if not container then return nil end
    for _, n in ipairs(names) do
        local v = container:GetAttribute(n)
        if v ~= nil then return v, n end
    end
    return nil
end

local function findValueObject(container, names)
    if not container then return nil end
    for _, n in ipairs(names) do
        local v = container:FindFirstChild(n)
        if v and (v:IsA("NumberValue") or v:IsA("IntValue")) then
            return v.Value, n, v
        end
    end
    return nil
end

function ArmorModule.get(targetPlayer)
    if not targetPlayer or not targetPlayer.Parent then return nil end
    local character = targetPlayer.Character
    if not character then return nil end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return nil end

    local armor = findAttribute(humanoid, ARMOR_ATTRIBUTES)
    if armor ~= nil then
        local maxArmor = findAttribute(humanoid, ARMOR_MAX_ATTRIBUTES) or DEFAULT_MAX_ARMOR
        return { percent = math.clamp((armor / maxArmor) * 100, 0, 100), current = armor, max = maxArmor }
    end

    armor = findAttribute(character, ARMOR_ATTRIBUTES)
    if armor ~= nil then
        local maxArmor = findAttribute(character, ARMOR_MAX_ATTRIBUTES) or DEFAULT_MAX_ARMOR
        return { percent = math.clamp((armor / maxArmor) * 100, 0, 100), current = armor, max = maxArmor }
    end

    for _, container in ipairs({humanoid, character}) do
        local value = findValueObject(container, ARMOR_VALUES)
        if value ~= nil then
            local maxValue = findValueObject(container, ARMOR_MAX_ATTRIBUTES)
            local max = maxValue or DEFAULT_MAX_ARMOR
            return { percent = math.clamp((value / max) * 100, 0, 100), current = value, max = max }
        end
    end

    return { percent = 0, current = 0, max = DEFAULT_MAX_ARMOR }
end

function ArmorModule.getColor(percent)
    if percent >= 75 then return Color3.fromRGB(90, 170, 255)
    elseif percent >= 50 then return Color3.fromRGB(60, 130, 230)
    elseif percent >= 25 then return Color3.fromRGB(45, 95, 200)
    else return Color3.fromRGB(30, 60, 160) end
end

-- ============================================================
-- HEALTH MODULE
-- ============================================================
local HealthModule = {}
function HealthModule.get(targetPlayer)
    if not targetPlayer or not targetPlayer.Parent then return nil end
    local character = targetPlayer.Character
    if not character then return nil end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return nil end
    local max = humanoid.MaxHealth
    if max <= 0 then return { percent = 0, current = 0, max = 100 } end
    return {
        percent = math.clamp((humanoid.Health / max) * 100, 0, 100),
        current = humanoid.Health,
        max = max
    }
end

function HealthModule.getColor(percent)
    if percent >= 75 then return Color3.fromRGB(60, 220, 100)
    elseif percent >= 50 then return Color3.fromRGB(255, 210, 70)
    elseif percent >= 25 then return Color3.fromRGB(255, 140, 50)
    else return Color3.fromRGB(230, 70, 70) end
end

-- ============================================================
-- SILENT AIM PREDICTION ENGINE
-- ============================================================
local SilentAimPrediction = (function()
    local M = {
        enabled       = false,
        hookInstalled = false,
        state = {
            target             = nil,
            isTargetting       = false,
            predicted_position = Vector3.new(1, 1, 1),
        },
        ghost_model = nil,
        ghost_owner = nil,
    }

    local CRISPY_YELLOW = Color3.fromRGB(255, 210, 60)
    local ORANGE        = Color3.fromRGB(255, 140, 0)

    local FOV_RADIUS_MULT = 1.5
    local PREDICTION      = 0.185
    local FOV_BASE        = 120
    M.fovBase = FOV_BASE
    M.wallCheck = false -- forced off -- removed UI; always off

    -- First solid hit must be the target (or empty). Anything else = wall.
    local function is_blocker_part(inst)
        if not inst or not inst:IsA("BasePart") then return false end
        if inst.CanCollide == false then return false end
        if inst.Transparency >= 0.95 then return false end
        if inst:FindFirstAncestorOfClass("Accessory") then return false end
        if inst:FindFirstAncestorOfClass("Tool") then return false end
        if inst.Name == "Handle" then return false end
        return true
    end

    -- Visible if Head OR HRP has clear line (ignores accessories / non-collide junk)
    local function is_visible(p)
        if not M.wallCheck then return true end
        if not p or not p.Character then return false end
        local char = p.Character
        local origin = camera.CFrame.Position
        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        local filter = {}
        if player.Character then table.insert(filter, player.Character) end
        params.FilterDescendantsInstances = filter
        params.IgnoreWater = true

        local function clear_to(part)
            if not part then return false end
            local delta = part.Position - origin
            local dist = delta.Magnitude
            if dist < 0.5 then return true end
            local hit = workspace:Raycast(origin, delta.Unit * (dist + 0.5), params)
            if not hit then return true end
            if hit.Instance:IsDescendantOf(char) then return true end
            -- ignore non-blocking clutter, keep casting is hard; treat non-blocker as clear-ish
            if not is_blocker_part(hit.Instance) then return true end
            return false
        end

        local head = char:FindFirstChild("Head")
        local hrp = char:FindFirstChild("HumanoidRootPart")
        return clear_to(head) or clear_to(hrp)
    end

    -- ---- Game support detection ----
    local game_support = {
        [2788229376]  = { Name = "Da Hood",              Remote = "MainEvent",  Argument = "UpdateMousePosI" },
        [12238627497] = { Name = "Locker Hood",          Remote = "MainEvent",  Argument = "UpdateMousePos" },
        [5602055394]  = { Name = "Hood Modded",          Remote = "MAINEVENT",  Argument = "MousePos" },
        [17403265390] = { Name = "Da Downhill",          Remote = "MAINEVENT",  Argument = "MOUSE" },
        [17403166075] = { Name = "Da Bank",              Remote = "MAINEVENT",  Argument = "MOUSE" },
        [18111448661] = { Name = "Da Uphill",            Remote = "MAINEVENT",  Argument = "MOUSE" },
        [15186202290] = { Name = "Da Strike",            Remote = "MAINEVENT",  Argument = "MOUSE" },
        [11143225577] = { Name = "1v1 Hood Aim Trainer", Remote = "MAINEVENT",  Argument = "UpdateMousePos" },
        [15763494605] = { Name = "Hood Aim",             Remote = "MAINEVENT",  Argument = "MOUSE" },
        [15166543806] = { Name = "Moon Hood",            Remote = "MAINEVENT",  Argument = "MoonUpdateMousePos" },
        [17897702920] = { Name = "OG Da Hood",           Remote = "MainEvent",  Argument = "UpdateMousePos" },
        [16033173781] = { Name = "Da Hood Macro",        Remote = "MainEvent",  Argument = "UpdateMousePos1" },
        [7213786345]  = { Name = "Da Hood VC",           Remote = "MainEvent",  Argument = "UpdateMousePosI" },
        [9825515356]  = { Name = "Hood Customs",         Remote = "MainEvent",  Argument = "MousePosUpdate" },
        [16859411452] = { Name = "Hood Z",               Remote = "MainEvent",  Argument = "UpdateMousePos" },
        [14277620939] = { Name = "Custom FFA",           Remote = "MainEvent",  Argument = "UpdateMousePos" },
    }

    local function auto_detect_remote()
        local candidates = {"MainEvent", "MAINEVENT", "Main", "Remote", "Remotes"}
        for _, name in ipairs(candidates) do
            local r = ReplicatedStorage:FindFirstChild(name)
            if r and r:IsA("RemoteEvent") then return r, name end
            local folder = ReplicatedStorage:FindFirstChild("Remotes")
            if folder then
                local r2 = folder:FindFirstChild(name)
                if r2 and r2:IsA("RemoteEvent") then return r2, name end
            end
        end
    end

    local game_config = game_support[game.PlaceId]
    local remote, mouse_argument
    if game_config then
        remote = ReplicatedStorage:FindFirstChild(game_config.Remote)
        mouse_argument = game_config.Argument
    else
        local detected, name = auto_detect_remote()
        if detected then
            remote = detected
            mouse_argument = "MOUSE"
            game_config = { Name = "Auto-Detected", Remote = name, Argument = mouse_argument }
        end
    end

    -- ---- Drawing objects (lazy) ----
    local drawings = nil

    local function ensureDrawings()
        if drawings then return end
        local ok, Drawing = pcall(function()
            return loadstring(game:HttpGet(
                "https://raw.githubusercontent.com/linemaster2/storage/main/Drawing.lua"
            ))()
        end)
        if not ok or not Drawing then
            warn("[Silent Aim Prediction] Failed to load Drawing library")
            return
        end

        local fov_outside = Drawing.new("Circle")
        fov_outside.Visible = false
        fov_outside.Filled = false
        fov_outside.Thickness = 1
        fov_outside.NumSides = 64

        local fov_inside = Drawing.new("Circle")
        fov_inside.Visible = false
        fov_inside.Filled = true
        fov_inside.NumSides = 64

        local tracer = Drawing.new("Line")
        tracer.Visible = false

        local center_dot = Drawing.new("Circle")
        center_dot.Visible = false
        center_dot.Filled = true
        center_dot.Radius = 4
        center_dot.Color = CRISPY_YELLOW
        center_dot.NumSides = 16

        local pm_outline = Drawing.new("Circle")
        pm_outline.Visible = false
        pm_outline.Filled = false
        pm_outline.Thickness = 2
        pm_outline.Color = ORANGE
        pm_outline.NumSides = 24

        local pm_fill = Drawing.new("Circle")
        pm_fill.Visible = false
        pm_fill.Filled = true
        pm_fill.Color = ORANGE
        pm_fill.Transparency = 0.2
        pm_fill.NumSides = 24

        drawings = {
            fov_outside = fov_outside,
            fov_inside  = fov_inside,
            tracer      = tracer,
            center_dot  = center_dot,
            pm_outline  = pm_outline,
            pm_fill     = pm_fill,
        }
    end

    -- ---- Helpers ----
    local function screen_center()
        local vp = camera.ViewportSize
        return Vector2.new(vp.X / 2, vp.Y / 2)
    end

    local function has_char(p)
        return p and p.Character and p.Character:FindFirstChild("Humanoid") ~= nil
    end

    local function world_to_screen(pos)
        local p, on = camera:WorldToViewportPoint(pos)
        return { position = Vector2.new(p.X, p.Y), on_screen = on }
    end

    local function is_in_air(p)
        if not has_char(p) then return false end
        local hrp = p.Character:FindFirstChild("HumanoidRootPart")
        return hrp and hrp.AssemblyLinearVelocity.Y ~= 0 or false
    end

    local function is_knocked(p)
        if not has_char(p) then return false end
        local be = p.Character:FindFirstChild("BodyEffects")
        if not be then return false end
        local ko = be:FindFirstChild("K.O")
        return ko and ko.Value or false
    end

    local function is_grabbed(p)
        if not has_char(p) then return false end
        return p.Character:FindFirstChild("GRABBING_CONSTRAINT") ~= nil
    end

    local function get_closest_player()
        local center = screen_center()
        local radius = (M.fovBase or FOV_BASE) * FOV_RADIUS_MULT
        local closest

        for _, p in ipairs(Players:GetPlayers()) do
            if p == player then continue end
            if not has_char(p) then continue end
            local hrp = p.Character:FindFirstChild("HumanoidRootPart")
            if not hrp then continue end
            local head = p.Character:FindFirstChild("Head")
            local screen = world_to_screen((head and head.Position) or hrp.Position)
            if not screen.on_screen then continue end
            if is_knocked(p) or is_grabbed(p) then continue end
            if not is_visible(p) then continue end
            local d = (center - screen.position).Magnitude
            if d <= radius then
                radius = d
                closest = p
            end
        end
        return closest
    end

    M.hitPart = "Head" -- Head | UpperTorso | HumanoidRootPart | LowerTorso
    M.autoPred = true
    M.manualPred = PREDICTION
    M.resolver = "None" -- None | Recalculate | MoveDirection
    M.smooth = 0 -- 0-1 lerp for visuals only; silent aim stays snaps

    local function get_aim_part(char)
        local name = M.hitPart or "Head"
        return char:FindFirstChild(name)
            or char:FindFirstChild("Head")
            or char:FindFirstChild("UpperTorso")
            or char:FindFirstChild("HumanoidRootPart")
    end

    local function resolve_velocity(p, hrp)
        local vel = hrp.AssemblyLinearVelocity
        if M.resolver == "MoveDirection" then
            local hum = p.Character and p.Character:FindFirstChildOfClass("Humanoid")
            if hum and hum.MoveDirection.Magnitude > 0.05 then
                local spd = math.max(vel.Magnitude, hum.WalkSpeed)
                vel = hum.MoveDirection.Unit * spd + Vector3.new(0, vel.Y, 0)
            end
        elseif M.resolver == "Recalculate" then
            -- blend assembly velocity with short history sample
            vel = Vector3.new(vel.X * 1.05, vel.Y * 0.3, vel.Z * 1.05)
        end
        return vel
    end

    local function get_predicted_position()
        local target = M.state.target
        if not target or not has_char(target) then return Vector3.new(0, 0, 0) end
        local char = target.Character
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if not hrp then return Vector3.new(0, 0, 0) end

        local aim_part = get_aim_part(char)
        if is_in_air(target) and (M.hitPart == "Head" or not M.hitPart) then
            aim_part = char:FindFirstChild("UpperTorso") or aim_part
        end

        local vel = resolve_velocity(target, hrp)
        local myRoot = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
        local dist = myRoot and (hrp.Position - myRoot.Position).Magnitude or 50

        local basePred = M.autoPred and PREDICTION or (tonumber(M.manualPred) or PREDICTION)

        -- CLOSE RANGE: almost no lead (over-prediction = misses point blank)
        local lead
        if dist < 8 then
            lead = 0.02
        elseif dist < 15 then
            lead = basePred * 0.08
        elseif dist < 28 then
            lead = basePred * 0.25
        elseif dist < 50 then
            lead = basePred * 0.55
        else
            lead = basePred * (1 + math.clamp(dist / 140, 0, 1.15))
        end

        local speed = Vector3.new(vel.X, 0, vel.Z).Magnitude
        if speed > 22 and dist > 35 then
            lead = lead * 1.15
        end

        -- point-blank: aim dead center of hit part
        if dist < 8 then
            return aim_part.Position
        end
        return aim_part.Position + Vector3.new(vel.X, vel.Y * 0.15, vel.Z) * lead
    end

    function M.setHitPart(name)
        M.hitPart = tostring(name or "Head")
    end
    function M.setPrediction(v)
        M.manualPred = tonumber(v) or M.manualPred
        PREDICTION = M.manualPred
        -- keep autoPred flag as set by Sync Lab; value always applied
    end
    function M.setAutoPred(v)
        M.autoPred = v and true or false
    end
    function M.setResolver(mode)
        M.resolver = mode or "None"
    end

    -- ---- Ghost clone ----
    local function cleanup_ghost()
        if M.ghost_model then
            M.ghost_model:Destroy()
            M.ghost_model = nil
            M.ghost_owner = nil
        end
    end

    -- Lightweight marker (NOT a full character clone — that caused FPS death)
    M.showGhost = false -- off by default for performance
    local function build_ghost(p)
        cleanup_ghost()
        if not M.showGhost then return end
        if not has_char(p) then return end
        local part = Instance.new("Part")
        part.Name = "SilentAimPredictionGhost"
        part.Size = Vector3.new(1.2, 1.2, 1.2)
        part.Shape = Enum.PartType.Ball
        part.Anchored = true
        part.CanCollide = false
        part.CanTouch = false
        part.CanQuery = false
        part.CastShadow = false
        part.Material = Enum.Material.Neon
        part.Color = ORANGE
        part.Transparency = 0.35
        part.Parent = workspace
        M.ghost_model = part
        M.ghost_owner = p
    end

    local function update_ghost(predicted_pos)
        if not M.showGhost then return end
        if not M.ghost_model or not M.state.target then return end
        if M.ghost_owner ~= M.state.target then
            build_ghost(M.state.target)
            if not M.ghost_model then return end
        end
        if M.ghost_model:IsA("BasePart") then
            M.ghost_model.Position = predicted_pos
        end
    end

    -- ---- __namecall hook (redirect all mouse-pos Vector3 fires) ----
    local MOUSE_ARGS = {
        UpdateMousePosI = true, UpdateMousePos = true, UpdateMousePos1 = true,
        MousePos = true, MOUSE = true, MousePosUpdate = true,
        MoonUpdateMousePos = true,
    }
    local function install_hook()
        if M.hookInstalled then return end
        if not _hookmetamethod then
            warn("[Silent Aim] hookmetamethod not available in this executor")
            return
        end
        M.hookInstalled = true
        local old
        old = _hookmetamethod(game, "__namecall", _newcclosure(function(self, ...)
            local method = _getnamecallmethod()
            local args = { ... }
            if not _checkcaller() and method == "FireServer" then
                if M.enabled and M.state.isTargetting and M.state.target then
                    local pred = M.state.predicted_position
                    -- string + Vector3 mouse updates
                    if typeof(args[1]) == "string" and typeof(args[2]) == "Vector3" then
                        if MOUSE_ARGS[args[1]] or args[1]:lower():find("mouse") then
                            args[2] = pred
                            return old(self, table.unpack(args))
                        end
                        -- still override unknown string+vec3 while targeting (Da Hood variants)
                        args[2] = pred
                        return old(self, table.unpack(args))
                    end
                    -- bare Vector3 first arg
                    if typeof(args[1]) == "Vector3" then
                        args[1] = pred
                        return old(self, table.unpack(args))
                    end
                end
            end
            return old(self, ...)
        end))
    end

    -- ---- Continuous remote feed (Da Hood needs mouse pos every frame while shooting) ----
    local function push_mouse_pos()
        if not (M.enabled and M.state.isTargetting and M.state.target) then return end
        local pred = M.state.predicted_position
        if not pred or pred.Magnitude < 1 then return end
        if remote and mouse_argument then
            pcall(function() remote:FireServer(mouse_argument, pred) end)
        end
        -- blanket common names on MainEvent / MAINEVENT
        for _, rname in ipairs({"MainEvent", "MAINEVENT"}) do
            local r = ReplicatedStorage:FindFirstChild(rname)
            if r then
                for _, arg in ipairs({"UpdateMousePosI", "UpdateMousePos", "MOUSE", "MousePos", "UpdateMousePos1"}) do
                    pcall(function() r:FireServer(arg, pred) end)
                end
            end
        end
    end

    -- ---- Render loop ----
    local _saAccum = 0
    local _saScanEvery = 0.03 -- ~33Hz target search
    local _saPushAccum = 0
    RunService.RenderStepped:Connect(function(dt)
        if not M.enabled then
            if drawings then
                for _, d in pairs(drawings) do d.Visible = false end
            end
            if M.state.target then
                M.state.target = nil
                M.state.isTargetting = false
                cleanup_ghost()
            end
            return
        end

        _saAccum = _saAccum + (dt or 0.016)
        if _saAccum >= _saScanEvery then
            _saAccum = 0
            local new_target = get_closest_player()
            if new_target ~= M.state.target then
                cleanup_ghost()
            end
            M.state.target = new_target
            M.state.isTargetting = new_target ~= nil
        end

        if M.state.isTargetting then
            -- drop target if wall check fails mid-lock
            if M.wallCheck and M.state.target and not is_visible(M.state.target) then
                M.state.target = nil
                M.state.isTargetting = false
                cleanup_ghost()
            else
                M.state.predicted_position = get_predicted_position()
                if M.showGhost then
                    if not M.ghost_model or M.ghost_owner ~= M.state.target then
                        build_ghost(M.state.target)
                    end
                    update_ghost(M.state.predicted_position)
                end
                -- keep server mouse locked on predicted head every frame
                _saPushAccum = _saPushAccum + (dt or 0.016)
                if _saPushAccum >= 0.016 then
                    _saPushAccum = 0
                    push_mouse_pos()
                end
            end
        else
            if M.ghost_model then cleanup_ghost() end
        end

        if not drawings then return end

        local center = screen_center()
        local radius = (M.fovBase or FOV_BASE) * FOV_RADIUS_MULT

        -- FOV outline only (skip filled circle — cheaper)
        drawings.fov_outside.Visible  = true
        drawings.fov_outside.Radius   = radius
        drawings.fov_outside.Color    = CRISPY_YELLOW
        drawings.fov_outside.Position = center

        drawings.fov_inside.Visible = false -- was a filled circle every frame (laggy)

        drawings.center_dot.Visible  = true
        drawings.center_dot.Position = center
        drawings.center_dot.Color    = CRISPY_YELLOW

        if M.state.isTargetting then
            local ps = world_to_screen(M.state.predicted_position)
            drawings.tracer.Visible      = ps.on_screen
            drawings.tracer.From         = center
            drawings.tracer.To           = ps.position
            drawings.tracer.Color        = CRISPY_YELLOW
            drawings.tracer.Thickness    = 1.5
            drawings.tracer.Transparency = 0.45

            drawings.pm_outline.Visible  = ps.on_screen
            drawings.pm_outline.Position = ps.position
            drawings.pm_outline.Radius   = 5
            drawings.pm_outline.Color    = ORANGE

            drawings.pm_fill.Visible = false -- skip fill
        else
            drawings.tracer.Visible     = false
            drawings.pm_outline.Visible = false
            drawings.pm_fill.Visible    = false
        end
    end)

    -- ---- Gun Activated — refresh predicted pos + spam mouse remotes ----
    local function fire_silent_shot()
        if not (M.enabled and M.state.isTargetting and M.state.target) then return end
        M.state.predicted_position = get_predicted_position()
        push_mouse_pos()
        -- double-tap next frame for packet timing
        task.defer(push_mouse_pos)
    end

    local function hook_gun(character)
        if not character then return end
        local function attach(tool)
            if not tool:IsA("Tool") then return end
            tool.Activated:Connect(function()
                fire_silent_shot()
            end)
        end
        for _, child in ipairs(character:GetChildren()) do
            attach(child)
        end
        character.ChildAdded:Connect(attach)
    end

    if player.Character then hook_gun(player.Character) end
    player.CharacterAdded:Connect(function(c)
        hook_gun(c)
        cleanup_ghost()
    end)

    -- ---- Public API ----
    function M.setEnabled(state)
        M.enabled = state and true or false
        if M.enabled then
            ensureDrawings()
            install_hook()
        else
            cleanup_ghost()
            M.state.target = nil
            M.state.isTargetting = false
            if drawings then
                for _, d in pairs(drawings) do d.Visible = false end
            end
        end
        print("[Silent Aim Prediction] " .. (M.enabled and "ON" or "OFF"))
    end

    function M.setFov(v)
        local n = tonumber(v) or 80
        M.fovBase = math.clamp(n, 20, 500)
        FOV_BASE = M.fovBase
    end

    function M.setWallCheck(v)
        M.wallCheck = false -- wall check removed
    end

    return M
end)()

    local targetHistory = {}
    local HISTORY_SIZE = 5
 
    local function getThumb(p)
        local ok, t = pcall(function()
            return Players:GetUserThumbnailAsync(p.UserId, Enum.ThumbnailType.HeadShot, Enum.ThumbnailSize.Size100x100)
        end)
        return ok and t or ""
    end
 
    local function getPlayersList()
        local list = {}
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= player then
                table.insert(list, p)
            end
        end
        table.sort(list, function(a, b)
            return (a.DisplayName or a.Name):lower() < (b.DisplayName or b.Name):lower()
        end)
        return list
    end
 
    local function getHeadAimPosition(tgtChar, tgtRoot)
        local head = tgtChar:FindFirstChild("Head")
        local aimPart = head or tgtRoot
        local vel = tgtRoot.AssemblyLinearVelocity
        local lead = 0.16
        local speed = Vector3.new(vel.X, 0, vel.Z).Magnitude
        if speed > 18 then lead = 0.22 end
        return aimPart.Position + Vector3.new(vel.X, vel.Y * 0.2, vel.Z) * lead
    end

    local function getRealTargetPosition(tgtRoot)
        local now = tick()
        local currentPos = tgtRoot.Position

        table.insert(targetHistory, {pos = currentPos, time = now})
        if #targetHistory > HISTORY_SIZE then
            table.remove(targetHistory, 1)
        end

        if #targetHistory >= 2 then
            local prev = targetHistory[#targetHistory - 1]
            local jump = (currentPos - prev.pos).Magnitude
            local dt = now - prev.time
            if jump > 30 and dt < 0.2 then
                return prev.pos
            end
        end

        local velocity = tgtRoot.AssemblyLinearVelocity
        if velocity.Magnitude > 1 then
            return currentPos + Vector3.new(velocity.X, velocity.Y * 0.2, velocity.Z) * 0.12
        end

        return currentPos
    end

    local function playKillAnim()
        local char = player.Character
        if not char then return end
        local humanoid = char:FindFirstChildOfClass("Humanoid")
        if not humanoid then return end
 
        if killAnimTrack then
            pcall(function() killAnimTrack:Stop() end)
            killAnimTrack = nil
        end
 
        local ok, anim = pcall(function()
            local a = Instance.new("Animation")
            a.AnimationId = killAnimId
            return humanoid:LoadAnimation(a)
        end)
        if not ok or not anim then return end
 
        killAnimTrack = anim
        killAnimTrack.Looped = true   -- â­ LOOPS FOREVER while Auto Kill is on
        pcall(function() killAnimTrack:Play() end)
    end
 
    local function stopKillAnim()
        if killAnimTrack then
            pcall(function() killAnimTrack:Stop() end)
            killAnimTrack = nil
        end
    end

        if autoKillEnabled then
            task.wait(0.5)

    local function equipTool(toolName)
        local char = player.Character
        if not char then return false end
        local humanoid = char:FindFirstChildOfClass("Humanoid")
        if not humanoid then return false end
        local equipped = char:FindFirstChild(toolName)
        if equipped and equipped:IsA("Tool") then return true end
        local backpack = player:FindFirstChild("Backpack")
        if backpack then
            local tool = backpack:FindFirstChild(toolName)
            if tool and tool:IsA("Tool") then
                humanoid:EquipTool(tool)
                return true
            end
        end
        return false
    end
 
    -- ============================================================
    -- AUTO SHOOT HELPER
    -- ============================================================
    local function pushTargetMouse(aimPos)
        if not aimPos then return end
        -- Feed silent aim engine so namecall hook + remotes lock onto target
        pcall(function()
            if SilentAimPrediction and SilentAimPrediction.state then
                SilentAimPrediction.state.target = selectedTarget
                SilentAimPrediction.state.isTargetting = true
                SilentAimPrediction.state.predicted_position = aimPos
            end
        end)
        for _, rname in ipairs({"MainEvent", "MAINEVENT"}) do
            local r = ReplicatedStorage:FindFirstChild(rname)
            if r then
                for _, arg in ipairs({
                    "UpdateMousePosI", "UpdateMousePos", "UpdateMousePos1",
                    "MOUSE", "MousePos", "MousePosUpdate", "MoonUpdateMousePos"
                }) do
                    pcall(function() r:FireServer(arg, aimPos) end)
                end
            end
        end
    end

    local function tryAutoShoot(aimPos)
        local char = player.Character
        if not char then return end
        local tool = char:FindFirstChildWhichIsA("Tool")
        if not tool then return end

        local toolName = tool.Name:lower()
        if toolName:find("katana") or toolName:find("knife")
            or toolName:find("sword") or toolName:find("bat")
            or toolName:find("fist") or toolName:find("crowbar") then
            return
        end

        if tick() - lastShot < AUTO_SHOOT_DELAY then return end
        lastShot = tick()

        if aimPos then
            pushTargetMouse(aimPos)
            task.defer(function() pushTargetMouse(aimPos) end)
        end
        pcall(function() tool:Activate() end)
    end

    local function runAutoKillTick()
        local t = selectedTarget
        if not t or not t.Parent or t == player then return end
        local myChar = player.Character
        local tgtChar = t.Character
        if not myChar or not tgtChar then return end
        local myRoot = myChar:FindFirstChild("HumanoidRootPart")
        local tgtRoot = tgtChar:FindFirstChild("HumanoidRootPart")
        local humanoid = myChar:FindFirstChildOfClass("Humanoid")
        if not myRoot or not tgtRoot or not humanoid then return end

        -- skip if target knocked
        local be = tgtChar:FindFirstChild("BodyEffects")
        if be then
            local ko = be:FindFirstChild("K.O")
            if ko and ko.Value then return end
        end

        if humanoid.Sit then humanoid.Sit = false end
        humanoid.PlatformStand = false
        humanoid.AutoRotate = false
        humanoid.WalkSpeed = 0
        humanoid.JumpPower = 0
        humanoid.JumpHeight = 0

        local aimPos = getHeadAimPosition(tgtChar, tgtRoot)
        local realTargetPos = getRealTargetPosition(tgtRoot)

        -- Stay close in FRONT of target (not random 15-stud sphere) so shots land
        local look = tgtRoot.CFrame.LookVector
        local side = tgtRoot.CFrame.RightVector
        local orbit = (tick() * 2.2) % (math.pi * 2)
        local dist = 7
        local newPos = realTargetPos
            - look * dist
            + side * (math.sin(orbit) * 2.5)
            + Vector3.new(0, 1.5, 0)

        -- Face the head
        myRoot.CFrame = CFrame.new(newPos, aimPos)
        myRoot.AssemblyLinearVelocity = Vector3.zero
        myRoot.AssemblyAngularVelocity = Vector3.zero

        -- Point camera at them too
        pcall(function()
            camera.CFrame = CFrame.new(camera.CFrame.Position, aimPos)
        end)

        tryAutoShoot(aimPos)

        local equipped = myChar:FindFirstChildWhichIsA("Tool")
        if not equipped then
            -- prefer any gun in backpack, else katana
            local bp = player:FindFirstChild("Backpack")
            local gun = nil
            if bp then
                for _, tool in ipairs(bp:GetChildren()) do
                    if tool:IsA("Tool") then
                        local n = tool.Name:lower()
                        if not (n:find("katana") or n:find("knife") or n:find("sword") or n:find("bat") or n:find("fist")) then
                            gun = tool
                            break
                        end
                    end
                end
            end
            if gun then
                pcall(function() humanoid:EquipTool(gun) end)
            else
                pcall(function() equipTool("Katana") end)
            end
        end
    end
 
    local function stopAutoKill()
        autoKillEnabled = false
        autoKillBtn.BackgroundColor3 = Color3.fromRGB(80, 60, 20)
        autoKillBtn.Text = "Auto Kill: OFF"
        if autoKillTask then
            task.cancel(autoKillTask)
            autoKillTask = nil
        end
        if pauseTask then
            task.cancel(pauseTask)
            pauseTask = nil
        end
        pauseTp = false
        targetHistory = {}
 
        stopKillAnim()
 
        local char = player.Character
        if char then
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum then
                hum.Sit = false
                hum.PlatformStand = false
                hum.AutoRotate = true
                hum.WalkSpeed = 16
                hum.JumpPower = 50
                hum.JumpHeight = 7.2
            end
        end
    end
 
    local function startAutoKill()
        autoKillEnabled = true
        pauseTp = false
        lastShot = 0
        targetHistory = {}
        autoKillBtn.BackgroundColor3 = Color3.fromRGB(80, 220, 120)
        autoKillBtn.Text = "Auto Kill: ON"
 
        playKillAnim()
 
        autoKillTask = task.spawn(function()
            while autoKillEnabled do
                if not pauseTp then
                    pcall(runAutoKillTick)
                end
                task.wait(0.05)
            end
        end)
 
        -- no periodic pause (was freezing auto-kill every second)
        pauseTask = nil
        pauseTp = false
    end
 
    autoKillBtn.MouseButton1Click:Connect(function()
        if autoKillEnabled then stopAutoKill() else startAutoKill() end
    end)
 
    task.spawn(function()
        while targetFrame.Parent do
            local t = selectedTarget or player
            local hInfo = HealthModule.get(t)
            if hInfo then
                local pct = hInfo.percent
                healthText.Text = string.format("Health  %d / %d", math.floor(hInfo.current), math.floor(hInfo.max))
                TweenService:Create(healthFill, TweenInfo.new(0.15), {
                    Size = UDim2.new(pct / 100, 0, 1, 0),
                    BackgroundColor3 = HealthModule.getColor(pct)
                }):Play()
            end
            local aInfo = ArmorModule.get(t)
            if aInfo then
                local pct = aInfo.percent
                armorText.Text = string.format("Armor  %d / %d", math.floor(aInfo.current), math.floor(aInfo.max))
                TweenService:Create(armorFill, TweenInfo.new(0.15), {
                    Size = UDim2.new(pct / 100, 0, 1, 0),
                    BackgroundColor3 = ArmorModule.getColor(pct)
                }):Play()
            end
            task.wait(0.15)
        end
    end)
 
    Players.PlayerAdded:Connect(function() refreshList() end)
    Players.PlayerRemoving:Connect(function(p)
        if selectedTarget == p then
            selectedTarget = player
            targetName.Text = "@" .. player.Name
            targetAvatar.Image = getThumb(player)
            targetHistory = {}
        end
        refreshList()
    end)
 
    return function()
        stopSpectate()
        stopAutoKill()
    end
end

-- Wire tabs EARLY so UI works even if a later module errors
local function switchTab(tabName)
    for name, frame in pairs(contentFrames) do
        frame.Visible = (name == tabName)
    end
    for name, btn in pairs(tabButtons) do
        if name == tabName then
            btn.BackgroundTransparency = 0.1
            btn.BackgroundColor3 = Color3.fromRGB(255, 220, 70)
        else
            btn.BackgroundTransparency = 0.35
            btn.BackgroundColor3 = Color3.fromRGB(255, 175, 40)
        end
    end
end
for name, btn in pairs(tabButtons) do
    btn.MouseButton1Click:Connect(function()
        pcall(switchTab, name)
    end)
    -- mobile
    btn.MouseButton1Up:Connect(function()
        pcall(switchTab, name)
    end)
end

    local function is_blocker_part(inst)
        if not inst or not inst:IsA("BasePart") then return false end
        if inst.CanCollide == false then return false end
        if inst.Transparency >= 0.95 then return false end

    local function is_player_behind_a_wall(p)
        if not has_character(player) or not has_character(p) then return true end
        local char = p.Character
        local origin = camera.CFrame.Position
        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        local filter = {}
        if player.Character then table.insert(filter, player.Character) end
        params.FilterDescendantsInstances = filter
        params.IgnoreWater = true

        local function blocked_to(part)
            if not part then return true end
            local delta = part.Position - origin
            local dist = delta.Magnitude
            if dist < 0.5 then return false end
            local hit = workspace:Raycast(origin, delta.Unit * (dist + 0.5), params)
            if not hit then return false end
            if hit.Instance:IsDescendantOf(char) then return false end
            if not is_blocker_part(hit.Instance) then return false end
            return true
        end

        local head = char:FindFirstChild("Head")
        local hrp = char:FindFirstChild("HumanoidRootPart")
        -- behind wall only if BOTH head and hrp are blocked
        return blocked_to(head) and blocked_to(hrp)
    end

        if not be then return false end

        return ko and ko.Value or false
    end

        if not has_character(p) then return false end
        return p.Character:FindFirstChild("[CarHitBox]") ~= nil
    end
 
    local function recalculate_velocity(part, update_time)
        local p0 = part.Position
        local t0 = tick()
        task.wait(1 / update_time)
        local p1 = part.Position
        local t1 = tick()
        local dt = t1 - t0
        if dt == 0 then return Vector3.new(0, 0, 0) end
        return (p1 - p0) / dt
    end
 
    local function get_player_in_look_direction()
        if not player.Character then return nil end
        local my_root = player.Character:FindFirstChild("HumanoidRootPart")
        if not my_root then return nil end
 
        local cam_cf = camera.CFrame
        local look_dir = cam_cf.LookVector
        local best_player = nil
        local best_dot = Settings.LookConeDot
 
        for _, p in ipairs(Players:GetPlayers()) do
            if p == player then continue end
            if not has_character(p) then continue end
            local head = p.Character:FindFirstChild("Head")
                or p.Character:FindFirstChild("HumanoidRootPart")
            if not head then continue end
 
            if Settings.UseChecks then
                if Settings.Checks.Knocked and is_knocked(p) then continue end
                if Settings.Checks.Grabbed and is_grabbed(p) then continue end
                if Settings.Checks.Friend and is_friends_with(p) then continue end
                if Settings.Checks.Wall and is_player_behind_a_wall(p) then continue end
                if Settings.Checks.Vehicle and is_on_vehicle(p) then continue end
            end
 
            local dir = (head.Position - cam_cf.Position)
            if dir.Magnitude == 0 then continue end
            dir = dir.Unit
            local dot = look_dir:Dot(dir)
            if dot > best_dot then
                best_dot = dot
                best_player = p
            end
        end
 
        return best_player
    end
 
    local function resolve_velocity(p)
        if not has_character(p) then return Vector3.new(0, 0, 0) end
        if Settings.ResolverMethod == "Recalculate" then
            return recalculate_velocity(p.Character.HumanoidRootPart, Settings.ResolverUpdateTime or 100)
        elseif Settings.ResolverMethod == "MoveDirection" then
            return p.Character.Humanoid.MoveDirection * p.Character.Humanoid.WalkSpeed
        end
        return Vector3.new(0, 0, 0)
    end
 
    local function get_aim_part(char)
        local n = Settings.HitPart or "Head"
        return char:FindFirstChild(n)
            or char:FindFirstChild("Head")
            or char:FindFirstChild("UpperTorso")
            or char:FindFirstChild("HumanoidRootPart")
    end

    local function get_predicted_position()
        local target = M.state.locked_target
        if not target or not has_character(target) then return Vector3.new(0, 0, 0) end
        local char = target.Character
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local aim_part = get_aim_part(char)
        if not aim_part then return Vector3.new(0, 0, 0) end

        local velocity = hrp and hrp.AssemblyLinearVelocity or aim_part.AssemblyLinearVelocity
        if Settings.ResolverEnabled and resolve_velocity then
            velocity = resolve_velocity(target) or velocity
        end
        -- MoveDirection resolver boost
        if SilentAimPrediction and SilentAimPrediction.resolver == "MoveDirection" then
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum and hum.MoveDirection.Magnitude > 0.05 then
                velocity = hum.MoveDirection.Unit * math.max(velocity.Magnitude, hum.WalkSpeed)
                    + Vector3.new(0, velocity.Y, 0)
            end
        end

        local myRoot = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
        local dist = (myRoot and hrp) and (hrp.Position - myRoot.Position).Magnitude or 50
        local pred = tonumber(Settings.Prediction) or 0.165
        if dist < 15 then pred = pred * 0.2
        elseif dist < 30 then pred = pred * 0.45 end

        if Settings.AntiGroundShots and is_in_air(target) then
            local mult = (Settings.AntiGroundToTakeOff or 5) / 10
            velocity = Vector3.new(velocity.X, math.abs(velocity.Y * mult), velocity.Z)
        end

        return aim_part.Position + Vector3.new(velocity.X, velocity.Y * 0.25, velocity.Z) * pred
    end

    local function do_lock()
        if not M.enabled then return end
        local target = get_player_in_look_direction()
        if target then
            M.state.locked_target = target
            M.state.isTargetting = true
            update_ui_locked()
            print("[Aimlock] LOCKED onto: " .. target.Name)
        else
            M.state.locked_target = nil
            M.state.isTargetting = false
            update_ui_unlocked()
        end
    end
 
    local function do_unlock()
        M.state.locked_target = nil
        M.state.isTargetting = false
        if dot_frame then dot_frame.Visible = false end
        update_ui_unlocked()
        print("[Aimlock] UNLOCKED")
    end
 
    function M.toggleLock()
        if not M.enabled then return end
        if M.state.locked_target then
            do_unlock()
        else
            do_lock()
        end
    end
 
    local function init()
        if M.initialized then return end
        M.initialized = true
 
        build_ui()
        update_ui_unlocked()
 
        if not _hookmetamethod then
            warn("[Aimlock] hookmetamethod not available")
            return
        end
        local __namecall
        __namecall = _hookmetamethod(game, "__namecall", _newcclosure(function(self, ...)
            local method = _getnamecallmethod()
            local args = { ... }
            if not _checkcaller() and method == "FireServer"
                and typeof(args[1]) == "string"
                and typeof(args[2]) == "Vector3" then
                if M.enabled and M.state.isTargetting and M.state.locked_target then
                    args[2] = M.state.predicted_position
                end
                return __namecall(self, table.unpack(args))
            end
            return __namecall(self, ...)
        end))
 
        local function hook_gun(character)
            if not character then return end
            character.ChildAdded:Connect(function(tool)
                if tool:IsA("Tool") then
                    tool.Activated:Connect(function()
                        if M.enabled and M.state.isTargetting and M.state.locked_target
                            and remote and mouse_argument then
                            pcall(function()
                                remote:FireServer(mouse_argument, M.state.predicted_position)
                            end)
                        end
                    end)
                end
            end)
        end
 
        if player.Character then hook_gun(player.Character) end
        player.CharacterAdded:Connect(function(c)
            hook_gun(c)
            do_unlock()
        end)
 
        UserInputService.InputBegan:Connect(function(input, gpe)
            if gpe then return end
            if input.KeyCode == Enum.KeyCode.E and M.enabled then
                M.toggleLock()
            end
        end)
 
        -- Run after default camera so lock is not overwritten
        RunService:BindToRenderStep("CrispyAimlockCam", Enum.RenderPriority.Camera.Value + 3, function()
            if not M.enabled then
                if dot_frame then dot_frame.Visible = false end
                return
            end

            if M.state.locked_target then
                if not M.state.locked_target.Parent
                    or not has_character(M.state.locked_target) then
                    do_unlock()
                end
            end
 
            if not M.state.locked_target then
                if dot_frame then dot_frame.Visible = false end
                return
            end
 
            M.state.isTargetting = true
            M.state.predicted_position = get_predicted_position()

            -- TRUE LOCK: CameraLock snaps camera AFTER Roblox camera step
            local pred = M.state.predicted_position
            local smooth = math.clamp(tonumber(Settings.Smooth) or 0, 0, 0.95)
            local method = Settings.Method or "CameraLock"
            if method == "CameraLock" then
                local cam = workspace.CurrentCamera
                if cam and pred and pred.Magnitude > 1 then
                    local goal = CFrame.new(cam.CFrame.Position, pred)
                    if smooth <= 0.05 then
                        cam.CFrame = goal
                    else
                        cam.CFrame = cam.CFrame:Lerp(goal, math.clamp(1 - smooth, 0.05, 1))
                    end
                end
            elseif method == "BodySnap" then
                local myChar = player.Character
                local root = myChar and myChar:FindFirstChild("HumanoidRootPart")
                if root and pred and pred.Magnitude > 1 then
                    local flat = Vector3.new(pred.X, root.Position.Y, pred.Z)
                    local goal = CFrame.new(root.Position, flat)
                    if smooth <= 0.05 then
                        root.CFrame = goal
                    else
                        root.CFrame = root.CFrame:Lerp(goal, math.clamp(1 - smooth, 0.05, 1))
                    end
                end
            end
 
            if Settings.ShowTargetDot and dot_frame then
                local head = M.state.locked_target.Character
                    and M.state.locked_target.Character:FindFirstChild("Head")
                if head then
                    local above = head.Position + Vector3.new(0, Settings.TargetDotOffset, 0)
                    local world_pos, on_screen = camera:WorldToViewportPoint(above)
                    if on_screen and world_pos.Z > 0 then
                        dot_frame.Visible = true
                        dot_frame.Position = UDim2.new(0, world_pos.X, 0, world_pos.Y)
                        dot_frame.BackgroundColor3 = Settings.TargetDotColor
                        dot_frame.BackgroundTransparency = Settings.TargetDotTransparency
                        dot_frame.Size = UDim2.new(0, Settings.TargetDotSize, 0, Settings.TargetDotSize)
                    else
                        dot_frame.Visible = false
                    end
                else
                    dot_frame.Visible = false
                end
            else
                if dot_frame then dot_frame.Visible = false end
            end
        end)
    end
 
    function M.setEnabled(state)
        M.enabled = state and true or false
        if M.enabled then
            init()
            if container then container.Visible = true end
            do_lock()
        else
            do_unlock()
            if container then container.Visible = false end
        end
        if M.onStateChange then
            pcall(M.onStateChange, M.enabled)
        end
        print("[Aimlock] " .. (M.enabled and "ON" or "OFF"))
    end

    function M.setWallCheck(v)
        Settings.Checks.Wall = false -- disabled
        print("[Aimlock] Wall Check " .. (Settings.Checks.Wall and "ON" or "OFF"))
    end
 
    -- Global keybind: Z toggles Aimlock on/off (with debounce)
    local zDebounce = false
    UserInputService.InputBegan:Connect(function(input, gpe)
        if gpe then return end
        if input.KeyCode ~= Enum.KeyCode.Z then return end
        if input.UserInputState ~= Enum.UserInputState.Begin then return end
        if zDebounce then return end
        zDebounce = true
        task.spawn(function()
            M.setEnabled(not M.enabled)
            task.wait(0.2)
            zDebounce = false
        end)
    end)
 
    return M
end)()

-- TRIGGERBOT MODULE (Da Strike — mobile-safe)
-- ============================================================
local TriggerBot = (function()
    local M = { enabled = false, initialized = false }

    local SETTINGS = {
        -- crosshair near player (not whole-screen FOV spam)
        AimRadius     = 20,   -- tight: must be near crosshair center
        FireDelay     = 0.08,
        Prediction    = 0.155,
        SilentAim     = true,
        VisibleCheck  = false, -- wall checks permanently off
        IgnoreKnocked = true,
        HitPart       = "Head",
    }

    local last_shot = 0
    local main_remote = ReplicatedStorage:FindFirstChild("MAINEVENT")
        or ReplicatedStorage:FindFirstChild("MainEvent")

    local function has_char(p)
        return p and p.Character and p.Character:FindFirstChild("HumanoidRootPart")
    end

    local function is_knocked(p)
        if not has_char(p) then return false end
        local be = p.Character:FindFirstChild("BodyEffects")
        if not be then return false end
        local ko = be:FindFirstChild("K.O")
        return ko and ko.Value or false
    end

    local function get_equipped_gun()
        if not player.Character then return nil end
        local tool = player.Character:FindFirstChildWhichIsA("Tool")
        if not tool then return nil end
        -- must look like a gun (has a Handle) — avoid fists / tools that aren't guns
        if not tool:FindFirstChild("Handle") then return nil end
        return tool
    end

    -- True only if the crosshair ray hits that player's character
    local function is_crosshair_on_player(p)
        if not has_char(p) then return false end
        local origin = camera.CFrame.Position
        local dir = camera.CFrame.LookVector * 500
        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        local filter = {}
        if player.Character then table.insert(filter, player.Character) end
        params.FilterDescendantsInstances = filter
        params.IgnoreWater = true
        local hit = workspace:Raycast(origin, dir, params)
        if not hit or not hit.Instance then return false end
        return hit.Instance:IsDescendantOf(p.Character)
    end

    local function is_blocker_part(inst)
        if not inst or not inst:IsA("BasePart") then return false end
        if inst.CanCollide == false then return false end
        if inst.Transparency >= 0.95 then return false end
        if inst:FindFirstAncestorOfClass("Accessory") then return false end
        if inst:FindFirstAncestorOfClass("Tool") then return false end
        return true
    end

    local function is_behind_wall(p)
        if not has_char(p) then return true end
        local char = p.Character
        local origin = camera.CFrame.Position
        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        local filter = {}
        if player.Character then table.insert(filter, player.Character) end
        params.FilterDescendantsInstances = filter
        params.IgnoreWater = true

        local function blocked_to(part)
            if not part then return true end
            local delta = part.Position - origin
            local dist = delta.Magnitude
            if dist < 0.5 then return false end
            local hit = workspace:Raycast(origin, delta.Unit * (dist + 0.5), params)
            if not hit then return false end
            if hit.Instance:IsDescendantOf(char) then return false end
            if not is_blocker_part(hit.Instance) then return false end
            return true
        end

        local head = char:FindFirstChild("Head")
        local hrp = char:FindFirstChild("HumanoidRootPart")
        return blocked_to(head) and blocked_to(hrp)
    end

    local GuiService = game:GetService("GuiService")

    -- Shiftlock / mouse-lock → screen center. Free mouse → real cursor position.
    local function get_aim_ray()
        local shiftOn = UserInputService.MouseBehavior == Enum.MouseBehavior.LockCenter
        if shiftOn then
            return camera.CFrame.Position, camera.CFrame.LookVector
        end
        -- free aim (PC mouse or mobile touch cursor)
        local mousePos = UserInputService:GetMouseLocation()
        local inset = GuiService:GetGuiInset()
        local x = mousePos.X - inset.X
        local y = mousePos.Y - inset.Y
        local ray = camera:ViewportPointToRay(x, y)
        return ray.Origin, ray.Direction
    end

    local function get_aim_screen_point()
        local shiftOn = UserInputService.MouseBehavior == Enum.MouseBehavior.LockCenter
        if shiftOn then
            return Vector2.new(camera.ViewportSize.X / 2, camera.ViewportSize.Y / 2)
        end
        local mousePos = UserInputService:GetMouseLocation()
        local inset = GuiService:GetGuiInset()
        return Vector2.new(mousePos.X - inset.X, mousePos.Y - inset.Y)
    end

    local function crosshair_hits_player(p)
        if not has_char(p) then return false end
        local origin, dir = get_aim_ray()
        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        local filter = {}
        if player.Character then table.insert(filter, player.Character) end
        params.FilterDescendantsInstances = filter
        params.IgnoreWater = true
        local hit = workspace:Raycast(origin, dir.Unit * 1000, params)
        if not hit or not hit.Instance then return false end
        return hit.Instance:IsDescendantOf(p.Character)
    end

    -- Only when cursor/crosshair is actually ON them (works shiftlock + free aim)
    local function get_aimed_player()
        local origin, dir = get_aim_ray()
        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        local filter = {}
        if player.Character then table.insert(filter, player.Character) end
        params.FilterDescendantsInstances = filter
        params.IgnoreWater = true

        local hit = workspace:Raycast(origin, dir.Unit * 1000, params)
        if hit and hit.Instance then
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= player and has_char(p) and hit.Instance:IsDescendantOf(p.Character) then
                    if SETTINGS.IgnoreKnocked and is_knocked(p) then return nil end
                    if SETTINGS.VisibleCheck and is_behind_wall(p) then return nil end
                    return p
                end
            end
        end

        -- tight fallback around real aim point (center if shiftlock, cursor if not)
        local aimScreen = get_aim_screen_point()
        local tight = math.min(SETTINGS.AimRadius, 22)
        local best, bestDist = nil, tight
        for _, p in ipairs(Players:GetPlayers()) do
            if p == player then continue end
            if not has_char(p) then continue end
            if SETTINGS.IgnoreKnocked and is_knocked(p) then continue end
            if SETTINGS.VisibleCheck and is_behind_wall(p) then continue end
            local char = p.Character
            for _, name in ipairs({"Head", "HumanoidRootPart"}) do
                local part = char:FindFirstChild(name)
                if part then
                    local sp, on = camera:WorldToViewportPoint(part.Position)
                    if on and sp.Z > 0 then
                        local d = (Vector2.new(sp.X, sp.Y) - aimScreen).Magnitude
                        if d < bestDist then
                            bestDist = d
                            best = p
                        end
                    end
                end
            end
        end
        if best and crosshair_hits_player(best) then
            return best
        end
        return nil
    end

    local function init()
        if M.initialized then return end
        M.initialized = true

        RunService.Heartbeat:Connect(function()
            if not M.enabled then return end
            if tick() - last_shot < SETTINGS.FireDelay then return end

            local gun = get_equipped_gun()
            if not gun then return end

            local target = get_aimed_player()
            if not target then return end -- NOT aiming at anyone → do not shoot

            local char = target.Character
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            local head = char and char:FindFirstChild("Head")
            if not hrp then return end
            local aimPos = (head and head.Position) or hrp.Position
            local vel = hrp.AssemblyLinearVelocity
            local predicted = aimPos + Vector3.new(vel.X, vel.Y * 0.35, vel.Z) * SETTINGS.Prediction

            last_shot = tick()

            if SETTINGS.SilentAim and main_remote then
                pcall(function() main_remote:FireServer("MOUSE", predicted) end)
                pcall(function() main_remote:FireServer("MousePos", predicted) end)
                pcall(function() main_remote:FireServer("UpdateMousePosI", predicted) end)
                pcall(function() main_remote:FireServer("UpdateMousePos", predicted) end)
            end

            -- also feed silent-aim prediction engine if running
            pcall(function()
                if SilentAimPrediction and SilentAimPrediction.state then
                    SilentAimPrediction.state.predicted_position = predicted
                    SilentAimPrediction.state.target = target
                    SilentAimPrediction.state.isTargetting = true
                end
            end)

            pcall(function()
                gun:Activate()
            end)
        end)

        print("[TriggerBot] Initialized (aim-only)")
    end

    function M.setEnabled(state)
        M.enabled = state and true or false
        if M.enabled then
            init()
            last_shot = 0
        end
        print("[TriggerBot] " .. (M.enabled and "ON" or "OFF"))
    end

    function M.setWallCheck(v)
        SETTINGS.VisibleCheck = v and true or false
        print("[TriggerBot] Wall Check " .. (SETTINGS.VisibleCheck and "ON" or "OFF"))
    end

    return M
end)()

    }
end

local function buildCombatTab(frame)
    local scroll = Instance.new("ScrollingFrame")
    scroll.Size = UDim2.new(1, 0, 1, 0)
    scroll.BackgroundTransparency = 1

    local saEnabled = false
    local saFOVRadius = 80
    local saFOVTransparency = 0.5
    local saState = { target = nil, isTargetting = false, predicted_position = Vector3.new(1, 1, 1) }
    local saInit = false
    local saVisuals = {}
 
    local SA_FOV_MULT = 1.5
    local SA_FOV_COLOR = Color3.fromRGB(255, 210, 60)
    local SA_PREDICTION = 0.134
 
    local function getScreenCenter()
        local vp = camera.ViewportSize
        return Vector2.new(vp.X / 2, vp.Y / 2)
    end
 
    local function worldToScreen(pos)
        local p, onScreen = camera:WorldToViewportPoint(pos)
        return { position = Vector2.new(p.X, p.Y), on_screen = onScreen }
    end
 
    local function hasChar(p)
        return (p and p.Character and p.Character:FindFirstChild("Humanoid")) and true or false
    end
 
    local function isInAir(p)
        if not hasChar(p) then return false end
        local root = p.Character:FindFirstChild("HumanoidRootPart")
        return root and root.Velocity.Y ~= 0 or false
    end
 
    local function isKnocked(p)
        if not hasChar(p) then return false end
        local be = p.Character:FindFirstChild("BodyEffects")
        if not be then return false end
        local ko = be:FindFirstChild("K.O")
        return ko and ko.Value or false
    end
 
    local function isGrabbed(p)
        if not hasChar(p) then return false end
        return p.Character:FindFirstChild("GRABBING_CONSTRAINT") ~= nil
    end
 
    local function getClosestPlayer()
        local center = getScreenCenter()
        local radius = saFOVRadius * SA_FOV_MULT
        local closest
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= player and hasChar(plr) then
                local root = plr.Character:FindFirstChild("HumanoidRootPart")
                if root then
                    local sp = worldToScreen(root.Position)
                    if sp.on_screen and not isKnocked(plr) and not isGrabbed(plr) then
                        local d = (center - sp.position).Magnitude
                        if d <= radius then
                            radius = d
                            closest = plr
                        end
                    end
                end
            end
        end
        return closest
    end
 
    local function getPredictedPosition()
        local target = saState.target
        if not target then return Vector3.new(0, 0, 0) end
        local root = target.Character:FindFirstChild("HumanoidRootPart")
        if not root then return Vector3.new(0, 0, 0) end
        local aimPart = root
        if isInAir(target) then
            local lt = target.Character:FindFirstChild("LowerTorso")
            if lt then aimPart = lt end
        end
        return aimPart.Position + root.Velocity * SA_PREDICTION
    end

        if saInit then return end
        saInit = true
 
        local Drawing = loadstring(game:HttpGet(

            fov_inside  = fov_inside,

        local oldNamecall
        oldNamecall = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
            local method = getnamecallmethod()
            local args = { ... }
            if not checkcaller() and method == "FireServer"
                and typeof(args[1]) == "string"
                and typeof(args[2]) == "Vector3" then
                if saEnabled and saState.isTargetting and saState.target then
                    args[2] = saState.predicted_position
                end
                return oldNamecall(self, table.unpack(args))
            end
            return oldNamecall(self, ...)
        end))

            if not saEnabled then
                saState.target = nil
                saState.isTargetting = false
                fov_outside.Visible = false
                fov_inside.Visible = false
                tracer.Visible = false
                center_dot.Visible = false

            end
 
            saState.target = getClosestPlayer()

-- FLY MODULE (Universal Fly v4 — camera-pitch aware)
-- ============================================================
local Fly = (function()
    local M = {
        enabled       = false,
        initialized   = false,
        speed         = 50,
        onStateChange = nil,
    }
 
    local CONFIG = {
        Speed           = 50,
        BoostMultiplier = 2.5,
        ToggleKey       = Enum.KeyCode.F,
        JumpHoldTime    = 0.15,
        MaxPitchFactor  = 1.0,
    }
 
    local IS_MOBILE = UserInputService.TouchEnabled
 
    local flying      = false
    local boost       = false
    local upHeld      = false
    local downHeld    = false
    local jumpUpExpire = 0
 
    local bodyVelocity
    local humanoid
    local rootPart
    local savedWalkSpeed, savedJumpPower, savedJumpHeight
 
    local function setJumpButtonVisible(visible)
        if not IS_MOBILE then return end
        local pg = player:FindFirstChild("PlayerGui")
        if not pg then return end
        local touchGui = pg:FindFirstChild("TouchGui")
        if not touchGui then return end
        local cf = touchGui:FindFirstChild("TouchControlFrame")
        if not cf then return end
        local jb = cf:FindFirstChild("JumpButton")
        if jb then jb.Visible = visible end
    end
 
    local function fireStateChange()
        if M.onStateChange then
            pcall(M.onStateChange, flying)
        end
    end
 
    local function startFly()
        if flying then return end
        local char = player.Character
        if not char then return end
 
        rootPart = char:FindFirstChild("HumanoidRootPart")
        humanoid = char:FindFirstChildOfClass("Humanoid")
        if not rootPart or not humanoid then return end
 
        savedWalkSpeed  = humanoid.WalkSpeed
        savedJumpPower  = humanoid.JumpPower
        savedJumpHeight = humanoid.JumpHeight
        humanoid.WalkSpeed  = 0
        humanoid.JumpPower  = 0
        humanoid.JumpHeight = 0
 
        bodyVelocity = Instance.new("BodyVelocity")
        bodyVelocity.MaxForce = Vector3.new(1e5, 1e5, 1e5)
        bodyVelocity.Velocity = Vector3.zero
        bodyVelocity.Parent   = rootPart
 
        flying = true
        boost  = false
        upHeld = false
        downHeld = false
 
        setJumpButtonVisible(false)
        fireStateChange()
    end
 
    local function stopFly()
        if not flying then return end
        flying = false
        boost  = false
        upHeld = false
        downHeld = false
 
        if bodyVelocity then
            bodyVelocity:Destroy()
            bodyVelocity = nil
        end
 
        if humanoid then
            humanoid.WalkSpeed  = savedWalkSpeed or 16
            humanoid.JumpPower  = savedJumpPower or 50
            humanoid.JumpHeight = savedJumpHeight or 7.2
        end
 
        setJumpButtonVisible(true)
        fireStateChange()
    end
 
    function M.toggle()
        if flying then stopFly() else startFly() end
    end
 
    function M.isOn()
        return flying
    end
 
    local function init()
        if M.initialized then return end
        M.initialized = true
 
        RunService.RenderStepped:Connect(function()
            if not flying then return end
 
            if not rootPart or not rootPart.Parent or not bodyVelocity or not humanoid then
                stopFly()
                return
            end
 
            local move = Vector3.zero
 
            local md = humanoid.MoveDirection
            if md.Magnitude > 0.01 then
                local localDir   = camera.CFrame:VectorToObjectSpace(md)
                local forwardAmt = -localDir.Z
                local rightAmt   =  localDir.X
 
                local look = camera.CFrame.LookVector
                if CONFIG.MaxPitchFactor < 1 then
                    local flat = Vector3.new(look.X, 0, look.Z)
                    if flat.Magnitude > 0 then
                        flat = flat.Unit
                        local pitch = math.asin(math.clamp(look.Y, -1, 1))
                        local maxP  = math.asin(CONFIG.MaxPitchFactor)
                        pitch = math.clamp(pitch, -maxP, maxP)
                        local horiz = math.cos(pitch)
                        look = Vector3.new(
                            flat.X * horiz,
                            math.sin(pitch),
                            flat.Z * horiz
                        ).Unit
                    end
                end
 
                move += look                      * forwardAmt
                move += camera.CFrame.RightVector * rightAmt
            end
 
            local now = tick()
            if upHeld or (now < jumpUpExpire) then move += Vector3.yAxis end
            if downHeld then move -= Vector3.yAxis end
 
            local speed = M.speed * (boost and CONFIG.BoostMultiplier or 1)
 
            if move.Magnitude > 0 then
                bodyVelocity.Velocity = move.Unit * speed
            else
                bodyVelocity.Velocity = Vector3.zero
            end
        end)
 
        UserInputService.InputBegan:Connect(function(input, gameProcessed)
            if gameProcessed then return end
            if input.UserInputType ~= Enum.UserInputType.Keyboard then return end
 
            local key = input.KeyCode
            if key == CONFIG.ToggleKey then
                M.toggle()
                return
            end
 
            if not flying then return end
 
            if key == Enum.KeyCode.Space or key == Enum.KeyCode.E then
                upHeld = true
            elseif key == Enum.KeyCode.LeftControl or key == Enum.KeyCode.Q then
                downHeld = true
            elseif key == Enum.KeyCode.LeftShift or key == Enum.KeyCode.RightShift then
                boost = true
            end
        end)
 
        UserInputService.InputEnded:Connect(function(input)
            if input.UserInputType ~= Enum.UserInputType.Keyboard then return end
            local key = input.KeyCode
            if key == Enum.KeyCode.Space or key == Enum.KeyCode.E then
                upHeld = false
            elseif key == Enum.KeyCode.LeftControl or key == Enum.KeyCode.Q then
                downHeld = false
            elseif key == Enum.KeyCode.LeftShift or key == Enum.KeyCode.RightShift then
                boost = false
            end
        end)
 
        UserInputService.JumpRequest:Connect(function()
            if flying and IS_MOBILE then
                jumpUpExpire = tick() + CONFIG.JumpHoldTime
            end
        end)
 
        player.CharacterAdded:Connect(function()
            if flying then stopFly() end
            rootPart = nil
            humanoid = nil
        end)
    end
 
    function M.setEnabled(state)
        if state then
            init()
            startFly()
        else
            stopFly()
        end
        print("[Fly] " .. (state and "ON" or "OFF"))
    end
 
    function M.setSpeed(v)
        M.speed = math.clamp(tonumber(v) or 50, 1, 100)
    end
 
    return M
end)()

-- ============================================================
local reloadEnabled = false
local lastAmmo = nil
local alreadyReloading = false
 
local function fireReload()
    if alreadyReloading then return end
    alreadyReloading = true
    pcall(function()
        local char = player.Character
        local backpack = player:FindFirstChild("Backpack")
        if char then
            local tool = char:FindFirstChildWhichIsA("Tool")
            if tool then
                local rl = tool:FindFirstChild("rl")
                if rl and rl:IsA("RemoteEvent") then rl:FireServer() end
            end
        end
        if backpack then
            for _, tool in ipairs(backpack:GetChildren()) do
                if tool:IsA("Tool") then
                    local rl = tool:FindFirstChild("rl")
                    if rl and rl:IsA("RemoteEvent") then rl:FireServer() end
                end
            end
        end
    end)
    task.wait(0.3)
    alreadyReloading = false
end
 
local function getCurrentAmmo()
    local char = player.Character
    if not char then return nil end
    local backpack = player:FindFirstChild("Backpack")
    local containers = {char}
    if backpack then table.insert(containers, backpack) end
    for _, container in ipairs(containers) do
        for _, tool in ipairs(container:GetChildren()) do
            if tool:IsA("Tool") then
                local ammo = tool:FindFirstChild("AMMO")
                if ammo and (ammo:IsA("NumberValue") or ammo:IsA("IntValue")) then
                    return ammo.Value
                end
            end
        end
    end
    return nil
end
 
local reloadToggle = createToggleRow("Auto Reload", 1, function(v)
    reloadEnabled = v
    lastAmmo = nil
end)
 
task.spawn(function()
    while screenGui.Parent do
        if reloadEnabled then
            local ammo = getCurrentAmmo()
            if ammo ~= nil then
                if ammo <= 0 and lastAmmo ~= nil and lastAmmo > 0 then
                    fireReload()
                end
                lastAmmo = ammo
            end
        else
            lastAmmo = nil
        end
        task.wait(0.1)
    end
end)

local AntiStomp = (function()
    local M = { enabled = false }
    local HP_THRESHOLD = 20
 
    local function forceReset(reason)
        local char = player.Character
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 then return end
        print("[Anti Stomp] " .. reason)
 
        pcall(function() char:BreakJoints() end)
        pcall(function() hum.Health = 0 end)
        pcall(function() hum:ChangeState(Enum.HumanoidStateType.Dead) end)
    end
 
    local function attach(character)
        if not character then return end
        local hum = character:WaitForChild("Humanoid", 5)
        if not hum then return end
 
        hum.HealthChanged:Connect(function(h)
            if not M.enabled then return end
            if h > 0 and h < HP_THRESHOLD then
                forceReset("health " .. math.floor(h) .. " < " .. HP_THRESHOLD)
            end
        end)
    end
 
    if player.Character then attach(player.Character) end
    player.CharacterAdded:Connect(attach)
 
    function M.setEnabled(state)
        M.enabled = state and true or false
        print("[Anti Stomp] " .. (M.enabled and "ON" or "OFF"))
    end
 
    return M
end)()

    local function find_attacker()
        local my_char = player.Character
        if not my_char then return nil end
        local my_hrp = my_char:FindFirstChild("HumanoidRootPart")
        if not my_hrp then return nil end
 
        local now = tick()
        local my_pos = my_hrp.Position
        local candidates = {}
 
        for _, p in ipairs(Players:GetPlayers()) do
            if p == player or not p.Character then continue end
            local hrp = p.Character:FindFirstChild("HumanoidRootPart")
            if not hrp then continue end
 
            local dist = (hrp.Position - my_pos).Magnitude
            if dist > CONFIG.MaxAttackerDistance then continue end
 
            local to_me = (my_pos - hrp.Position)
            if to_me.Magnitude < 0.01 then continue end
            to_me = to_me.Unit
            local facing_dot = to_me:Dot(hrp.CFrame.LookVector)
 
            local shot = recent_shots[p]
            local shot_age = shot and (now - shot.time) or 999
            local recently_shot = shot_age <= CONFIG.RecentShotWindow
 
            local trajectory_score = 0
            if shot and shot.direction then
                local to_us_from_origin = (my_pos - shot.origin)
                if to_us_from_origin.Magnitude > 0.1 then
                    to_us_from_origin = to_us_from_origin.Unit
                    local dir_match = shot.direction:Dot(to_us_from_origin)
                    if dir_match > 0.95 then
                        trajectory_score = 100
                    elseif dir_match > 0.85 then
                        trajectory_score = 70
                    elseif dir_match > 0.7 then
                        trajectory_score = 30
                    end
                end
            end
 
            local has_los = has_line_of_sight(hrp.Position, my_pos, my_char, p.Character)
 
            local score = 0
            if facing_dot >= CONFIG.FacingThreshold then
                score += facing_dot * 60
            end
            if recently_shot and shot then
                local recency_bonus = math.max(0, 80 * (1 - shot_age / CONFIG.RecentShotWindow))
                score += recency_bonus
                score += shot.confidence * 30
            end
            score += trajectory_score
            if has_los then score += 20 else score -= 30 end
            score -= math.min(dist * 0.08, 25)
 
            if facing_dot >= CONFIG.FacingThreshold or recently_shot or trajectory_score > 0 then
                table.insert(candidates, {
                    player = p,
                    score = score,
                })
            end
        end
 
        table.sort(candidates, function(a, b) return a.score > b.score end)
 
        local top = candidates[1]
        if top and top.score >= CONFIG.MinConfidenceScore then
            return top.player
        end
 
        return nil
    end
 
    local function on_hit_detected()
        if not M.enabled then return end
        local now = tick()
        if now - last_hit_time < CONFIG.DebounceWindow then return end
        last_hit_time = now
 
        local attacker = find_attacker()
        if attacker then
            show_hit(attacker.DisplayName)
        else
            show_hit("Someone")
        end
    end
 
    local function is_real_hit_blood(obj, character)
        if obj.Name ~= "BloodParticle"
            and obj.Name ~= "BloodSplatter"
            and obj.Name ~= "BloodParticles" then
            return false
        end
 
        local parent = obj.Parent
        if not parent then return false end
 
        local valid = false
        for _, name in ipairs(HIT_PARENTS) do
            if parent.Name == name then valid = true break end
        end
        if not valid then return false end
 
        local p = parent
        while p and p ~= character do
            if p:IsA("Tool") then return false end
            p = p.Parent
        end
 
        return true
    end
 
    local function watch_character(character)
        local humanoid = character:WaitForChild("Humanoid", 5)
        local last_health = humanoid and humanoid.Health or 100
 
        character.DescendantAdded:Connect(function(obj)
            if is_real_hit_blood(obj, character) then
                on_hit_detected()
            end
        end)
 
        if humanoid then
            humanoid.HealthChanged:Connect(function(new_health)
                if not M.enabled then
                    last_health = new_health
                    return
                end
                local drop = last_health - new_health
                if drop >= 1 and tick() - last_hit_time > 0.05 then
                    on_hit_detected()
                end
                last_health = new_health
            end)
        end
    end
 
    -- ---- Init ----
    function M.init()
        if M.initialized then return end
        M.initialized = true
 
        screen_gui = Instance.new("ScreenGui")
        screen_gui.Name = "CrispyHitNotifs"
        screen_gui.ResetOnSpawn = false
        screen_gui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
        screen_gui.Parent = CoreGui
 
        active_notifs = {}
        merge_map = {}
 
        workspace.ChildAdded:Connect(function(obj)
            if not M.enabled then return end
            if not is_bullet(obj) then return end
 
            task.spawn(function()
                local origin = obj.Position
 
                local beam, end_pos
                for i = 1, 25 do
                    beam = find_beam(obj)
                    if beam and beam.Attachment1 then
                        local wp = beam.Attachment1.WorldPosition
                        if wp and wp.Magnitude > 0 then
                            end_pos = wp
                            break
                        end
                    end
                    task.wait(0.01)
                end
 
                local direction
                if end_pos then
                    local diff = end_pos - origin
                    if diff.Magnitude > 0.1 then
                        direction = diff.Unit
                    end
                end
 
                local candidates = {}
                for _, p in ipairs(Players:GetPlayers()) do
                    if p == player or not p.Character then continue end
                    local hrp = p.Character:FindFirstChild("HumanoidRootPart")
                    if not hrp then continue end
                    local d = (hrp.Position - origin).Magnitude
                    if d <= CONFIG.BulletOwnerRadius then
                        table.insert(candidates, { player = p, distance = d })
                    end
                end
 
                if #candidates == 1 then
                    local p = candidates[1].player
                    recent_shots[p] = {
                        time = tick(),
                        origin = origin,
                        direction = direction,
                        confidence = 1.0,
                    }
                elseif #candidates > 1 then
                    table.sort(candidates, function(a, b) return a.distance < b.distance end)
                    local p = candidates[1].player
                    recent_shots[p] = {
                        time = tick(),
                        origin = origin,
                        direction = direction,
                        confidence = 0.4,
                    }
                end
            end)
        end)
 
        if player.Character then watch_character(player.Character) end
        player.CharacterAdded:Connect(watch_character)
 
        print("[Hit Detection] Initialized")
    end
 
    function M.setEnabled(state)
        M.enabled = state and true or false
        if M.enabled then
            M.init()
        else
            if active_notifs then
                for _, c in ipairs(active_notifs) do
                    if c and c.Parent then c:Destroy() end
                end
                active_notifs = {}
                merge_map = {}
                recent_shots = {}
            end
        end
        print("[Hit Detection] " .. (M.enabled and "ON" or "OFF"))
    end
 
    return M
end)()

local function removeCharacterEffects(character)
    if not character then return end
 
    -- 1. Disable + clear the particle FIRST so nothing new spawns
    local hrp = character:FindFirstChild("HumanoidRootPart")
    if hrp then
        local attach = hrp:FindFirstChild("AuraAttachment")
        if attach then
            local particle = attach:FindFirstChild("AuraParticle")
            if particle then
                particle.Enabled = false
                pcall(function() particle:Clear() end)
            end
            attach:Destroy()
        end
    end
 
    -- 2. Sweep for any leftover attachments / particle emitters
    for _, d in ipairs(character:GetDescendants()) do
        if d.Name == "AuraAttachment" then
            d:Destroy()
        elseif d:IsA("ParticleEmitter") and d.Name == "AuraParticle" then
            d:Destroy()
        end
    end
 
    -- 3. Destroy the effects folder (hat, etc.)
    local oldFolder = character:FindFirstChild("OPEffects")
    if oldFolder then oldFolder:Destroy() end
 
    -- 4. Sweep highlights — but leave ESP chams alone
    for _, d in ipairs(character:GetDescendants()) do
        if d:IsA("Highlight") and d.Name ~= "YellowESP_Chams" then
            d:Destroy()
        end
    end
    local topHL = character:FindFirstChildOfClass("Highlight")
    if topHL and topHL.Name ~= "YellowESP_Chams" then topHL:Destroy() end
end

local ESP = (function()
    local M = { enabled = false, initialized = false }
 
    local CFG = {
        ChamsFill      = Color3.fromRGB(255, 240, 40),
        ChamsOutline   = Color3.fromRGB(255, 160, 0),
        ChamsFillAlpha = 0.55,
        ChamsOutAlpha  = 0,
        ChamsName      = "YellowESP_Chams",
    }
 
    local function ensureChams(character)
        local hl = character:FindFirstChild(CFG.ChamsName)
        if hl then return hl end
 
        hl = Instance.new("Highlight")
        hl.Name                = CFG.ChamsName
        hl.Adornee             = character
        hl.FillColor           = CFG.ChamsFill
        hl.OutlineColor        = CFG.ChamsOutline
        hl.FillTransparency    = 1 - CFG.ChamsFillAlpha
        hl.OutlineTransparency = CFG.ChamsOutAlpha
        hl.DepthMode           = Enum.HighlightDepthMode.AlwaysOnTop
        hl.Parent              = character
        return hl
    end
 
    local function destroyChams(character)
        if not character then return end
        local hl = character:FindFirstChild(CFG.ChamsName)
        if hl then hl:Destroy() end
    end
 
    local function removePlayer(p)
        if p.Character then destroyChams(p.Character) end
    end
 
    local function clearAll()
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= player and p.Character then
                destroyChams(p.Character)
            end
        end
    end
 
    local function init()
        if M.initialized then return end
        M.initialized = true
 
        RunService.RenderStepped:Connect(function()
            if not M.enabled then return end
 
            for _, p in ipairs(Players:GetPlayers()) do
                if p == player then continue end
 
                local char = p.Character
                local hum  = char and char:FindFirstChildOfClass("Humanoid")
 
                if not (char and hum) or hum.Health <= 0 then
                    if char then destroyChams(char) end
                    continue
                end
 
                ensureChams(char)
            end
        end)
 
        Players.PlayerRemoving:Connect(removePlayer)
    end
 
    function M.setEnabled(state)
        M.enabled = state and true or false
        if M.enabled then
            init()
        else
            clearAll()
        end
        print("[ESP] " .. (M.enabled and "ON" or "OFF"))
    end
 
    return M
end)()

-- clamp any frame on screen
local function clampToScreen(frame)
    local cam = workspace.CurrentCamera
    if not cam then return end
    local vs = cam.ViewportSize
    local abs = frame.AbsolutePosition
    local size = frame.AbsoluteSize
    local x = math.clamp(abs.X, 0, math.max(0, vs.X - size.X))
    local y = math.clamp(abs.Y, 0, math.max(0, vs.Y - size.Y))
    frame.Position = UDim2.fromOffset(x, y)
end

    end)
end

local function keyName(k)
    if not k or k == Enum.KeyCode.Unknown then return "None" end
    return k.Name
end

local function fireToggle(name)
    if name == "SilentAim" then
        SilentAimPrediction.setEnabled(not SilentAimPrediction.enabled)
    elseif name == "Aimlock" then
        Aimlock.setEnabled(not Aimlock.enabled)
    elseif name == "TriggerBot" then
        TriggerBot.setEnabled(not TriggerBot.enabled)
    end
    -- refresh side button colors

UserInputService.InputBegan:Connect(function(input, gp)
    if waitingBind then
        if input.UserInputType == Enum.UserInputType.Keyboard
            or input.UserInputType == Enum.UserInputType.Gamepad1 then
            if input.KeyCode == Enum.KeyCode.Escape then
                waitingBind = nil
                return
            end
            if input.KeyCode ~= Enum.KeyCode.Unknown then
                Keybinds[waitingBind] = input.KeyCode
                if KeybindLabels[waitingBind] then
                    KeybindLabels[waitingBind].Text = keyName(input.KeyCode)
                end
                waitingBind = nil
                playClick()
            end
        end
        return
    end
    if gp then return end
    if input.UserInputType == Enum.UserInputType.Keyboard
        or input.UserInputType == Enum.UserInputType.Gamepad1 then

local function makeSideDraggable(frame)
    local draggingS, start, startPos, moved = false, nil, nil, false
    frame.InputBegan:Connect(function(inp)
        if Side.locked then return end -- locked = no drag
        if inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch then
            draggingS = true
            moved = false
            start = inp.Position
            startPos = frame.Position
            local c
            c = inp.Changed:Connect(function()
                if inp.UserInputState == Enum.UserInputState.End then
                    draggingS = false
                    clampToScreen(frame)
                    if c then c:Disconnect() end
                end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(inp)
        if Side.locked or not draggingS then return end
        if inp.UserInputType == Enum.UserInputType.MouseMovement or inp.UserInputType == Enum.UserInputType.Touch then
            local d = inp.Position - start
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
end

local function ensureConfigFolder()
    if not isfolder then return end
    pcall(function()
        if not isfolder(CONFIG_FOLDER) then makefolder(CONFIG_FOLDER) end
    end)
end

local function saveConfig()
    ensureConfigFolder()
    if not writefile then return false end
    -- pull live values
    pcall(function()
        if SilentAimPrediction then
            CrispyConfig.silentHitPart = SilentAimPrediction.hitPart or "Head"
            CrispyConfig.prediction = SilentAimPrediction.manualPred or CrispyConfig.prediction
            CrispyConfig.autoPred = SilentAimPrediction.autoPred
            CrispyConfig.resolver = SilentAimPrediction.resolver or "None"
        end
        if Aimlock and Aimlock.Settings then
            CrispyConfig.aimHitPart = Aimlock.Settings.HitPart or "Head"
            CrispyConfig.smooth = Aimlock.Settings.Smooth or 0
            CrispyConfig.aimMethod = Aimlock.Settings.Method or "CameraLock"
            if Aimlock.Settings.Prediction then CrispyConfig.prediction = Aimlock.Settings.Prediction end
        end
        if Side then CrispyConfig.sideLocked = Side.locked end
    end)
    local ok = pcall(function()
        writefile(CONFIG_FILE, HttpService:JSONEncode(CrispyConfig))
    end)
    return ok
end

local function loadConfig()
    if not (isfile and readfile) then return end
    pcall(function()
        if not isfile(CONFIG_FILE) then return end
        local data = HttpService:JSONDecode(readfile(CONFIG_FILE))
        if type(data) ~= "table" then return end
        for k, v in pairs(data) do CrispyConfig[k] = v end
        if SilentAimPrediction then
            SilentAimPrediction.hitPart = CrispyConfig.silentHitPart
            SilentAimPrediction.manualPred = CrispyConfig.prediction
            SilentAimPrediction.autoPred = CrispyConfig.autoPred
            SilentAimPrediction.resolver = CrispyConfig.resolver
            if SilentAimPrediction.setPrediction and not CrispyConfig.autoPred then
                SilentAimPrediction.setPrediction(CrispyConfig.prediction)
            end
        end
        if Aimlock and Aimlock.Settings then
            Aimlock.Settings.HitPart = CrispyConfig.aimHitPart
            Aimlock.Settings.Smooth = CrispyConfig.smooth
            Aimlock.Settings.Method = CrispyConfig.aimMethod
            Aimlock.Settings.Prediction = CrispyConfig.prediction
        end
        if Side then Side.locked = CrispyConfig.sideLocked ~= false end
    end)
end
pcall(loadConfig)

local function getPingMs()
    local ping = 60
    pcall(function()
        local net = Stats:FindFirstChild("Network")
        local item = net and net:FindFirstChild("ServerStatsItem") and net.ServerStatsItem:FindFirstChild("Data Ping")
        if item and item.GetValue then
            local v = item:GetValue()
            if typeof(v) == "number" then
                ping = v
            elseif typeof(v) == "string" then
                ping = tonumber((v:gsub("ms", ""):gsub("%s", ""))) or ping
            end
        end
    end)
    -- fallback: PerformanceStats if available
    pcall(function()
        if ping == 60 and Stats.GetTotalMemoryUsageMb then
            -- keep default
        end
    end)
    return math.clamp(tonumber(ping) or 60, 1, 600)
end

-- Prediction lead scales with one-way latency (ping/2 in seconds-ish for Hood)
local function bestPredForPing(ms)
    ms = tonumber(ms) or 60
    if ms <= 35 then return 0.07
    elseif ms <= 50 then return 0.10
    elseif ms <= 70 then return 0.13
    elseif ms <= 90 then return 0.155
    elseif ms <= 110 then return 0.18
    elseif ms <= 130 then return 0.205
    elseif ms <= 150 then return 0.23
    elseif ms <= 180 then return 0.26
    elseif ms <= 210 then return 0.29
    elseif ms <= 250 then return 0.33
    elseif ms <= 300 then return 0.38
    elseif ms <= 350 then return 0.42
    elseif ms <= 400 then return 0.47
    else return 0.52 end -- 400ms+
end

local function applyPred(val, auto)
    val = math.clamp(tonumber(val) or 0.16, 0.01, 0.60)
    CrispyConfig.prediction = val
    CrispyConfig.autoPred = auto and true or false
    pcall(function()
        -- Silent Aim engine
        if SilentAimPrediction then
            SilentAimPrediction.manualPred = val
            SilentAimPrediction.autoPred = true -- always use live value
            if SilentAimPrediction.setPrediction then
                SilentAimPrediction.setPrediction(val)
            end
        end
        -- Aimlock / camera lock
        if Aimlock and Aimlock.Settings then
            Aimlock.Settings.Prediction = val
        end
        -- TriggerBot if present
        if TriggerBot and TriggerBot.SETTINGS then
            TriggerBot.SETTINGS.Prediction = val
        end
    end)
    print(string.format("[Sync Lab] Pred=%.3f auto=%s (SA + Aimlock)", val, tostring(CrispyConfig.autoPred)))
end

-- Movement extras: speed, noclip, inf jump, FOV, hitbox, config
do
    local frame = contentFrames["Movement"]
    if frame then
        local holder = frame:FindFirstChildWhichIsA("ScrollingFrame")
        holder = holder and holder:FindFirstChildWhichIsA("Frame") or frame
        local orderBase = 50
        neonLabel("—— EXTRA ——", orderBase, holder)

        -- Speed
        local speedOn = CrispyConfig.speedOn
        local speedVal = CrispyConfig.speedVal or 24
        local speedBtn = neonBtn("Speed: OFF", orderBase + 1, holder, function(b)
            speedOn = not speedOn
            CrispyConfig.speedOn = speedOn
            b.Text = speedOn and ("Speed: ON (" .. speedVal .. ")") or "Speed: OFF"
            saveConfig()
        end)
        neonBtn("Speed −5", orderBase + 2, holder, function()
            speedVal = math.max(1, speedVal - 5)
            CrispyConfig.speedVal = speedVal
            if speedOn then speedBtn.Text = "Speed: ON (" .. speedVal .. ")" end
            saveConfig()
        end)
        neonBtn("Speed +5", orderBase + 3, holder, function()
            speedVal = math.min(100, speedVal + 5)
            CrispyConfig.speedVal = speedVal
            if speedOn then speedBtn.Text = "Speed: ON (" .. speedVal .. ")" end
            saveConfig()
        end)
        RunService.Heartbeat:Connect(function()
            if not speedOn then return end
            local char = player.Character
            if not char then return end
            local hum = char:FindFirstChildOfClass("Humanoid")
            local root = char:FindFirstChild("HumanoidRootPart")
            if not hum or not root or hum.Health <= 0 then return end
            pcall(function() hum.WalkSpeed = speedVal end)
            local dir = hum.MoveDirection
            if dir.Magnitude > 0.05 then
                local v = root.AssemblyLinearVelocity
                root.AssemblyLinearVelocity = Vector3.new(dir.X * speedVal, v.Y, dir.Z * speedVal)
            end
        end)

        -- Noclip
        local noclipOn = false
        neonBtn("Noclip: OFF", orderBase + 4, holder, function(b)
            noclipOn = not noclipOn
            CrispyConfig.noclip = noclipOn
            b.Text = noclipOn and "Noclip: ON" or "Noclip: OFF"
            saveConfig()
        end)
        RunService.Stepped:Connect(function()
            if not noclipOn then return end
            local char = player.Character
            if not char then return end
            for _, p in ipairs(char:GetDescendants()) do
                if p:IsA("BasePart") then
                    p.CanCollide = false
                end
            end
        end)

        -- No jump cooldown + Inf jump
        local noJumpCd = false
        local infJump = false
        local noJumpBtn2 = neonBtn("No Jump Cooldown: OFF", orderBase + 5, holder, function(b)
            noJumpCd = not noJumpCd
            CrispyConfig.noJumpCd = noJumpCd
            b.Text = noJumpCd and "No Jump Cooldown: ON" or "No Jump Cooldown: OFF"
            if noJumpCd then
                pcall(function()
                    loadstring(game:HttpGet("https://pastebin.com/raw/3hmrC7bc"))()
                end)
            end
            saveConfig()
        end)
        neonBtn("Inf Jump: OFF", orderBase + 6, holder, function(b)
            infJump = not infJump
            CrispyConfig.infJump = infJump
            b.Text = infJump and "Inf Jump: ON" or "Inf Jump: OFF"
            if infJump and not noJumpCd then
                noJumpCd = true
                CrispyConfig.noJumpCd = true
                noJumpBtn2.Text = "No Jump Cooldown: ON"
                pcall(function()
                    loadstring(game:HttpGet("https://pastebin.com/raw/3hmrC7bc"))()
                end)
            end
            if not infJump then
                noJumpCd = false
                CrispyConfig.noJumpCd = false
                noJumpBtn2.Text = "No Jump Cooldown: OFF"
            end
            saveConfig()
        end)
        UserInputService.JumpRequest:Connect(function()
            if not infJump then return end
            local char = player.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            if hum then
                hum:ChangeState(Enum.HumanoidStateType.Jumping)
            end
        end)

        -- FOV
        local fovVal = CrispyConfig.fov or 70
        local fovBtn = neonBtn("FOV: " .. tostring(fovVal), orderBase + 7, holder, function() end)
        neonBtn("FOV −5", orderBase + 8, holder, function()
            fovVal = math.max(50, fovVal - 5)
            CrispyConfig.fov = fovVal
            pcall(function() workspace.CurrentCamera.FieldOfView = fovVal end)
            fovBtn.Text = "FOV: " .. fovVal
            saveConfig()
        end)
        neonBtn("FOV +5 (max 120)", orderBase + 9, holder, function()
            fovVal = math.min(120, fovVal + 5)
            CrispyConfig.fov = fovVal
            pcall(function() workspace.CurrentCamera.FieldOfView = fovVal end)
            fovBtn.Text = "FOV: " .. fovVal
            saveConfig()
        end)

        -- Hitbox expander
        local hitboxOn = false
        neonBtn("Hitbox Expander: OFF", orderBase + 10, holder, function(b)
            hitboxOn = not hitboxOn
            CrispyConfig.hitbox = hitboxOn
            b.Text = hitboxOn and "Hitbox Expander: ON" or "Hitbox Expander: OFF"
            saveConfig()
        end)
        RunService.Heartbeat:Connect(function()
            if not hitboxOn then return end
            local size = CrispyConfig.hitboxSize or 2.5
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr ~= player and plr.Character then
                    local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
                    if hrp then
                        pcall(function()
                            hrp.Size = Vector3.new(size, size, size)
                            hrp.Transparency = 0.7
                            hrp.CanCollide = false
                        end)
                    end
                end
            end
        end)

        neonBtn("Save Config", orderBase + 20, holder, function(b)
            if saveConfig() then b.Text = "Saved!" else b.Text = "Save failed (no writefile)" end
            task.delay(1, function() b.Text = "Save Config" end)
        end)
        neonBtn("Delete Config", orderBase + 21, holder, function(b)
            pcall(function()
                if isfile and isfile(CONFIG_FILE) and delfile then delfile(CONFIG_FILE) end
            end)
            b.Text = "Deleted"
            task.delay(1, function() b.Text = "Delete Config" end)
        end)
    end
end

    overlay.BorderSizePixel = 0

    Instance.new("UICorner", overlay).CornerRadius = UDim.new(0, 10)
    local ost = Instance.new("UIStroke", overlay)
    ost.Color = Color3.fromRGB(255, 220, 80)
    ost.Thickness = 1.5
    local opad = Instance.new("UIPadding", overlay)
    opad.PaddingTop = UDim.new(0, 6)
    opad.PaddingBottom = UDim.new(0, 6)
    opad.PaddingLeft = UDim.new(0, 8)
    opad.PaddingRight = UDim.new(0, 8)
    local olayout = Instance.new("UIListLayout", overlay)
    olayout.Padding = UDim.new(0, 2)
    olayout.SortOrder = Enum.SortOrder.LayoutOrder

    local titleO = Instance.new("TextLabel")
    titleO.Size = UDim2.new(1, 0, 0, 18)
    titleO.BackgroundTransparency = 1
    titleO.Text = "ACTIVE"
    titleO.TextColor3 = Color3.fromRGB(40, 25, 5)
    titleO.Font = Enum.Font.GothamBlack
    titleO.TextSize = 12
    titleO.LayoutOrder = 0
    titleO.Parent = overlay

    local lines = {}
    local function line(order)
        local l = Instance.new("TextLabel")
        l.Size = UDim2.new(1, 0, 0, 14)
        l.BackgroundTransparency = 1
        l.Text = ""
        l.TextColor3 = Color3.fromRGB(40, 25, 5)
        l.Font = Enum.Font.GothamBold
        l.TextSize = 11
        l.TextXAlignment = Enum.TextXAlignment.Left
        l.LayoutOrder = order
        l.Parent = overlay
        lines[#lines + 1] = l
        return l
    end
    for i = 1, 10 do line(i) end

    local function refreshOverlay()
        local active = {}
        pcall(function()
            if SilentAimPrediction and SilentAimPrediction.enabled then active[#active+1] = "Silent Aim" end
            if Aimlock and Aimlock.enabled then active[#active+1] = "Aimlock" end
            if TriggerBot and TriggerBot.enabled then active[#active+1] = "TriggerBot" end
            if CrispyConfig and CrispyConfig.speedOn then active[#active+1] = "Speed " .. tostring(CrispyConfig.speedVal or "") end
            if CrispyConfig and CrispyConfig.noclip then active[#active+1] = "Noclip" end
            if CrispyConfig and CrispyConfig.infJump then active[#active+1] = "Inf Jump" end
            if CrispyConfig and CrispyConfig.noJumpCd then active[#active+1] = "No Jump CD" end
            if CrispyConfig and CrispyConfig.hitbox then active[#active+1] = "Hitbox" end
            if Fly and Fly.isOn and Fly.isOn() then active[#active+1] = "Fly" end
            if Side and Side.show then active[#active+1] = "Side Btns" end
        end)
        if #active == 0 then
            lines[1].Text = "None"
            lines[1].TextColor3 = Color3.fromRGB(100, 70, 30)
            for i = 2, #lines do lines[i].Text = "" end
            overlay.Visible = true
            return
        end
        for i, l in ipairs(lines) do
            if active[i] then
                l.Text = "• " .. active[i]
                l.TextColor3 = Color3.fromRGB(20, 120, 40)
            else
                l.Text = ""
            end
        end
        overlay.Visible = true
    end
    task.spawn(function()
        while overlay.Parent do
            refreshOverlay()
            task.wait(0.4)
        end
    end)
end
