                    State.autostomp.isteleported = true
                end
                if Utils.hascharacter(target3) then

                    ReplicatedStorage.MainEvent:FireServer("Stomp")
                end)

                    ReplicatedStorage.MAINEVENT:FireServer("STOMP")
                end)

            return
        end
