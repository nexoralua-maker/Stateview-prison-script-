--save manager
local SaveManager = {}
local SM = { Library = Library, Folder = "crispy.gg/game", Ignore = {}, AutoloadFile = "autoload.txt" }
local FS = Library.FileSystem
function SaveManager:SetLibrary(lib) SM.Library = lib end
function SaveManager:SetFolder(folder) SM.Folder = folder; Library:BuildFolders(folder) end
function SaveManager:SetIgnoreIndexes(list) SM.Ignore = list or {} end
function SaveManager:IgnoreThemeSettings() end
function SaveManager:GetConfigPath(name) return SM.Folder .. "/configs/" .. tostring(name) .. ".json" end
function SaveManager:GetAutoloadPath() return SM.Folder .. "/settings/" .. SM.AutoloadFile end
function SaveManager:Save(name)
    if not name or name == "" then return false end
    local data = SM.Library:GetConfig(SM.Ignore)
    local ok, encoded = pcall(function() return game:GetService("HttpService"):JSONEncode(data) end)
    if not ok then return false end
    Library:BuildFolders(SM.Folder); Library:BuildFolders(SM.Folder .. "/configs")
    FS:Write(self:GetConfigPath(name), encoded); return true
end
function SaveManager:Load(name)
    if not name or name == "" then return false end
    local raw = FS:Read(self:GetConfigPath(name)); if not raw then return false end
    local ok, data = pcall(function() return game:GetService("HttpService"):JSONDecode(raw) end)
    if not ok or type(data) ~= "table" then return false end
    SM.Library:LoadConfig(data, SM.Ignore); return true
end
function SaveManager:Delete(name) if not name or name == "" then return false end; FS:Delete(self:GetConfigPath(name)); return true end
function SaveManager:List()
    local files = FS:List(SM.Folder .. "/configs"); local out = {}
    for _, f in ipairs(files) do local nm = string.match(f, "^(.+)%.json$"); if nm then table.insert(out, nm) end end
    table.sort(out); return out
end
function SaveManager:RefreshConfigList() end
function SaveManager:GetConfigList() return self:List() end
function SaveManager:SetAutoloadConfig(name)
    Library:BuildFolders(SM.Folder); Library:BuildFolders(SM.Folder .. "/settings")
    FS:Write(self:GetAutoloadPath(), tostring(name or "")); return true
end
function SaveManager:GetAutoload()
    local raw = FS:Read(self:GetAutoloadPath()); if not raw or raw == "" then return nil end
    return raw
end
function SaveManager:ClearAutoload() return self:SetAutoloadConfig("") end
function SaveManager:LoadAutoloadConfig()
    local name = self:GetAutoload(); if not name or name == "" then return false end
    return self:Load(name)
end

--notify helper
local function notify(title, text, kind)
    pcall(function()
        if kind == "success" and Library.Success then
            Library:Success(title, text, 3)
        elseif kind == "warn" and Library.Warn then
            Library:Warn(title, text, 3)
        elseif kind == "error" and Library.Error then
            Library:Error(title, text, 3)
        else
            Library:Notify({ Title = title, Text = text, Duration = 3 })
        end
    end)
end

local CLOSEST_PART_NAMES = {"Head","Neck","UpperTorso","LowerTorso","HumanoidRootPart","LeftUpperArm","RightUpperArm","LeftLowerArm","RightLowerArm","LeftHand","RightHand","LeftUpperLeg","RightUpperLeg","LeftLowerLeg","RightLowerLeg","LeftFoot","RightFoot"}
local function ResolveClosestPart(character, requestedPart)
    if not character then return nil end
    if requestedPart ~= "Closest Part" then
        return character:FindFirstChild(requestedPart) or character:FindFirstChild("HumanoidRootPart")
    end
    local camPos = Camera.CFrame.Position
    local best, bestDist = nil, math.huge
    for _, name in ipairs(CLOSEST_PART_NAMES) do
        local part = character:FindFirstChild(name)
        if part then
            local d = (part.Position - camPos).Magnitude
            if d < bestDist then bestDist = d; best = part end
        end
    end
    return best or character:FindFirstChild("HumanoidRootPart")
end

local Aimlock = (function()

        lockKey = Enum.KeyCode.Insert,
        state = { locked_target = nil, isTargetting = false, predicted_position = Vector3.new(1,1,1) },

        autoAir = true, autoAirDelay = 0.2,
    }
    local lastAutoAir = 0
    local Settings = {
        HPrediction = 0.134, VPrediction = 0.134,
        AimPart = "Head", MaxDistance = 500, LookConeDot = 0.90,
        UseChecks = true,
        Checks = { Knocked = true, Grabbed = true, Friend = false, Wall = false, Vehicle = false },
        AntiGroundShots = true, AntiGroundMult = 0.2,
        ShowTargetDot = true,
        TargetDotColor = WHITE, TargetDotColor2 = WHITE,
        TargetDotSize = 12, TargetDotOffset = 1.2,
    }
    local game_support = {
        [2788229376]={Remote="MainEvent",Argument="UpdateMousePosI"},[12238627497]={Remote="MainEvent",Argument="UpdateMousePos"},
        [5602055394]={Remote="MAINEVENT",Argument="MousePos"},[17403265390]={Remote="MAINEVENT",Argument="MOUSE"},
        [17403166075]={Remote="MAINEVENT",Argument="MOUSE"},[18111448661]={Remote="MAINEVENT",Argument="MOUSE"},
        [15186202290]={Remote="MAINEVENT",Argument="MOUSE"},[11143225577]={Remote="MAINEVENT",Argument="UpdateMousePos"},
        [15763494605]={Remote="MAINEVENT",Argument="MOUSE"},[15166543806]={Remote="MAINEVENT",Argument="MoonUpdateMousePos"},
        [17897702920]={Remote="MainEvent",Argument="UpdateMousePos"},[16033173781]={Remote="MainEvent",Argument="UpdateMousePos1"},
        [7213786345]={Remote="MainEvent",Argument="UpdateMousePosI"},[9825515356]={Remote="MainEvent",Argument="MousePosUpdate"},
        [16859411452]={Remote="MainEvent",Argument="UpdateMousePos"},[14277620939]={Remote="MainEvent",Argument="UpdateMousePos"},
    }
    local remote, mouse_argument
    do
        local cfg = game_support[game.PlaceId]
        if cfg then remote = ReplicatedStorage:FindFirstChild(cfg.Remote); mouse_argument = cfg.Argument
        else
            for _, name in ipairs({"MainEvent","MAINEVENT","Main","Remote"}) do
                local r = ReplicatedStorage:FindFirstChild(name)
                if r and r:IsA("RemoteEvent") then remote = r; mouse_argument = "MOUSE"; break end
            end
        end
    end
    local function has_char(p) return p and p.Character and p.Character:FindFirstChild("Humanoid") ~= nil end
    local function is_knocked(p) if not has_char(p) then return false end; local be = p.Character:FindFirstChild("BodyEffects"); if not be then return false end; local ko = be:FindFirstChild("K.O"); return ko and ko.Value or false end
    local function is_grabbed(p) if not has_char(p) then return false end; return p.Character:FindFirstChild("GRABBING_CONSTRAINT") ~= nil end
    local function is_visible(targetChar, targetPart)
        if not targetChar or not targetPart then return false end
        local origin = Camera.CFrame.Position; local dir = targetPart.Position - origin
        if dir.Magnitude < 0.1 then return true end
        local params = RaycastParams.new()
        local ok = pcall(function() params.FilterType = Enum.RaycastFilterType.Exclude end)
        if not ok then pcall(function() params.FilterType = Enum.RaycastFilterType.Blacklist end) end
        local exclude = { Camera, targetChar }
        if LP.Character then table.insert(exclude, LP.Character); for _, d in ipairs(LP.Character:GetDescendants()) do table.insert(exclude, d) end end
        for _, d in ipairs(targetChar:GetDescendants()) do table.insert(exclude, d) end
        params.FilterDescendantsInstances = exclude; params.IgnoreWater = true
        local result = Workspace:Raycast(origin, dir, params)
        if not result then return true end
        return result.Instance and result.Instance:IsDescendantOf(targetChar) or false
    end
    local function is_in_air(p)
        if not has_char(p) then return false end
        local root = p.Character:FindFirstChild("HumanoidRootPart")
        return root and math.abs(root.AssemblyLinearVelocity.Y) > 1 or false
    end
    local function get_target_in_look_direction()
        if not has_char(LP) then return nil end
        local myRoot = LP.Character:FindFirstChild("HumanoidRootPart"); if not myRoot then return nil end
        local camCF, lookDir = Camera.CFrame, Camera.CFrame.LookVector
        local best, bestDot = nil, Settings.LookConeDot
        for _, p in ipairs(Players:GetPlayers()) do
            if p == LP then continue end
            if not has_char(p) then continue end
            local head = ResolveClosestPart(p.Character, Settings.AimPart); if not head then continue end
            if (head.Position - myRoot.Position).Magnitude > Settings.MaxDistance then continue end
            if Settings.UseChecks then
                if Settings.Checks.Knocked and is_knocked(p) then continue end
                if Settings.Checks.Grabbed and is_grabbed(p) then continue end
                if Settings.Checks.Friend  and p:IsFriendsWith(LP.UserId) then continue end
                if Settings.Checks.Vehicle and p.Character:FindFirstChild("[CarHitBox]") then continue end
            end
            if Settings.Checks.Wall and not is_visible(p.Character, head) then continue end
            local dir = head.Position - camCF.Position
            if dir.Magnitude < 0.01 then continue end
            local dot = lookDir:Dot(dir.Unit)
            if dot > bestDot then bestDot = dot; best = p end
        end
        return best
    end
    local function get_predicted_position()
        local target = M.state.locked_target
        if not has_char(target) then return Vector3.new(0, 0, 0) end
        local aimPart = ResolveClosestPart(target.Character, Settings.AimPart)
        if not aimPart then return Vector3.new(0, 0, 0) end
        local pos, vel = aimPart.Position, aimPart.AssemblyLinearVelocity
        if Settings.AntiGroundShots and is_in_air(target) then
            vel = Vector3.new(vel.X, math.abs(vel.Y * Settings.AntiGroundMult), vel.Z)
        end
        return Vector3.new(pos.X + vel.X * Settings.HPrediction, pos.Y + vel.Y * Settings.VPrediction, pos.Z)
    end
    local dot_frame
    local function ensure_dot()
        if dot_frame and dot_frame.Parent then return end
        local gui = LP:FindFirstChild("PlayerGui"); if not gui then return end
        local sg = gui:FindFirstChild("DaLockGui")
        if not sg then
            sg = Instance.new("ScreenGui"); sg.Name = "DaLockGui"; sg.ResetOnSpawn = false
            sg.IgnoreGuiInset = true; sg.DisplayOrder = 999; sg.Parent = gui
        end
        dot_frame = Instance.new("Frame")
        dot_frame.AnchorPoint = Vector2.new(0.5, 0.5)
        dot_frame.Size = UDim2.new(0, Settings.TargetDotSize, 0, Settings.TargetDotSize)
        dot_frame.BackgroundColor3 = WHITE; dot_frame.BorderSizePixel = 0
        dot_frame.Visible = false; dot_frame.ZIndex = 50; dot_frame.Parent = sg
        Instance.new("UICorner", dot_frame).CornerRadius = UDim.new(1, 0)
        local gradient = Instance.new("UIGradient")
        gradient.Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0.00, Settings.TargetDotColor2),
            ColorSequenceKeypoint.new(0.50, Settings.TargetDotColor),
            ColorSequenceKeypoint.new(1.00, Settings.TargetDotColor),
        })
        gradient.Rotation = 45; gradient.Parent = dot_frame; dot_gradient_ref = gradient
    end
    local function do_lock()
        if not M.enabled then return end
        local target = get_target_in_look_direction()
        if target then
            M.state.locked_target = target
            M.state.isTargetting = true
            if M.onLockChange then pcall(M.onLockChange, true, target) end
        else
            M.state.locked_target = nil
            M.state.isTargetting = false
            if M.onLockChange then pcall(M.onLockChange, false, nil) end
        end
    end
    local function do_unlock()
        local had = M.state.locked_target ~= nil
        M.state.locked_target = nil
        M.state.isTargetting = false
        if dot_frame then dot_frame.Visible = false end
        if had and M.onLockChange then pcall(M.onLockChange, false, nil) end
    end
    function M.toggleLock() if not M.enabled then return end; if M.state.locked_target then do_unlock() else do_lock() end end
    local function install_hook()
        local old
        old = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
            local method = getnamecallmethod(); local args = { ... }
            if not checkcaller() and method == "FireServer" and typeof(args[1]) == "string" and typeof(args[2]) == "Vector3" then
                if M.enabled and M.state.isTargetting and M.state.locked_target then args[2] = M.state.predicted_position end
                return old(self, table.unpack(args))
            end
            return old(self, ...)
        end))
    end
    local function hook_gun(character)
        if not character then return end
        character.ChildAdded:Connect(function(tool)
            if not tool:IsA("Tool") then return end
            tool.Activated:Connect(function()
                if M.enabled and M.state.isTargetting and M.state.locked_target and remote and mouse_argument then
                    pcall(function() remote:FireServer(mouse_argument, M.state.predicted_position) end)
                end
            end)
        end)
    end
    local function init()
        if M.initialized then return end
        M.initialized = true; ensure_dot(); install_hook()
        if LP.Character then hook_gun(LP.Character) end
        LP.CharacterAdded:Connect(hook_gun)
        UIS.InputBegan:Connect(function(input, gpe)
            if gpe then return end
            if input.UserInputState ~= Enum.UserInputState.Begin then return end
            if input.KeyCode == M.lockKey then M.toggleLock() end
        end)
        RunService.RenderStepped:Connect(function()
            if not M.enabled then if dot_frame then dot_frame.Visible = false end; return end
            if M.state.locked_target and not M.state.locked_target.Parent then do_unlock() end
            if M.state.locked_target and not has_char(M.state.locked_target) then do_unlock() end
            if not M.state.locked_target then if dot_frame then dot_frame.Visible = false end; return end
            M.state.isTargetting = true; M.state.predicted_position = get_predicted_position()
            if M.camlock then
                local goal = CFrame.new(Camera.CFrame.Position, M.state.predicted_position)
                if M.camSmooth then Camera.CFrame = Camera.CFrame:Lerp(goal, M.camSmoothValue)
                else Camera.CFrame = goal end
            end
        end)
        pcall(function()
            RunService:BindToRenderStep("CrispyAimlockDot", Enum.RenderPriority.Camera.Value + 1, function()
                if not M.enabled or not dot_frame then if dot_frame then dot_frame.Visible = false end; return end
                if not M.state.locked_target or not has_char(M.state.locked_target) then dot_frame.Visible = false; return end
                if Settings.ShowTargetDot then
                    local head = ResolveClosestPart(M.state.locked_target.Character, Settings.AimPart)
                    if head then
                        local above = head.Position + Vector3.new(0, Settings.TargetDotOffset, 0)
                        local sp, on = Camera:WorldToViewportPoint(above)
                        if on and sp.Z > 0 then dot_frame.Visible = true; dot_frame.Position = UDim2.new(0, sp.X, 0, sp.Y)
                        else dot_frame.Visible = false end
                    else dot_frame.Visible = false end
                end
            end)
        end)
        RunService.Heartbeat:Connect(function()
            if not M.enabled or not M.autoAir then return end
            local t = M.state.locked_target
            if not has_char(t) then return end
            local hum = t.Character:FindFirstChildOfClass("Humanoid")
            if not hum then return end
            local ok, st = pcall(function() return hum:GetState() end)
            local airborne = ok and (st == Enum.HumanoidStateType.Jumping or st == Enum.HumanoidStateType.Freefall or hum.FloorMaterial == Enum.Material.Air)
            if not airborne then return end
            if tick() - lastAutoAir < M.autoAirDelay then return end
            lastAutoAir = tick()
            local char = LP.Character; if not char then return end
            local tool = char:FindFirstChildOfClass("Tool")
            if tool then pcall(function() tool:Activate() end) end
        end)
    end
    function M.setEnabled(state)
        M.enabled = state and true or false
        if M.enabled then init(); do_lock() else do_unlock() end
        if M.onStateChange then pcall(M.onStateChange, M.enabled) end
    end
    function M.setHPrediction(v) Settings.HPrediction = tonumber(v) or Settings.HPrediction end
    function M.setVPrediction(v) Settings.VPrediction = tonumber(v) or Settings.VPrediction end
    function M.setAimPart(part) Settings.AimPart = part end
    function M.setWallCheck(v) Settings.Checks.Wall = v and true or false end
    function M.setCamlock(v) M.camlock = v and true or false end
    function M.setCamSmooth(v) M.camSmooth = v and true or false end
    function M.setCamSmoothValue(v) M.camSmoothValue = math.clamp(tonumber(v) or M.camSmoothValue, 0.01, 1) end
    function M.setAutoAir(v) M.autoAir = v and true or false end
    function M.setAutoAirDelay(v) M.autoAirDelay = tonumber(v) or M.autoAirDelay end
    M.Settings = Settings
    return M
end)()

local SilentAimPrediction = (function()

        state = { target=nil, isTargetting=false, predicted_position=Vector3.new(1,1,1) },
        ghost_model=nil, ghost_owner=nil }
    local FOV_RADIUS_MULT, H_PREDICTION, V_PREDICTION = 1.5, 0.134, 0.134
    local FOV_BASE, FOV_TRANSPARENCY = 80, 0.5
    local AIM_PART, WALLCHECK = "Head", false

--auto fire
local AutoFire = (function()
    local M = { enabled=false, initialized=false }
    local FIRE_DELAY, IGNORE_MELEE = 0.1, true
    local activeTool, lastFireTime, loopConn = nil, 0, nil
    local function is_melee_or_consumable(tool)
        if not IGNORE_MELEE then return false end
        local n = tool.Name:lower()
        return n:find("katana") or n:find("knife") or n:find("sword") or n:find("bat") or n:find("fist") or n:find("crowbar") or n:find("pizza") or n:find("donut") or n:find("cola")
    end
    local function is_gun(tool) if not tool or not tool:IsA("Tool") then return false end; if not tool:FindFirstChild("Handle") then return false end; if is_melee_or_consumable(tool) then return false end; return true end
    local function activateTool(tool) if not tool or not tool.Parent then return end; if tool.Parent ~= player.Character then return end; pcall(function() tool:Activate() end) end
    local function onEquip(tool)
        if not M.enabled then return end; if not is_gun(tool) then return end
        activeTool, lastFireTime = tool, 0; task.wait(0.05)
        if M.enabled and activeTool == tool and tool.Parent == player.Character then activateTool(tool); lastFireTime = tick() end
    end
    local function hookCharacter(character)
        if not character then return end
        character.ChildAdded:Connect(function(child) if child:IsA("Tool") then onEquip(child) end end)
        character.ChildRemoved:Connect(function(child) if child:IsA("Tool") and child == activeTool then activeTool = nil end end)
    end
    local function startLoop()
        if loopConn then return end
        loopConn = RunService.Heartbeat:Connect(function()
            if not M.enabled then return end
            if not activeTool or activeTool.Parent ~= player.Character then return end
            if tick() - lastFireTime >= FIRE_DELAY then activateTool(activeTool); lastFireTime = tick() end
        end)
    end
    local function init()
        if M.initialized then return end; M.initialized = true
        if player.Character then hookCharacter(player.Character); local already = player.Character:FindFirstChildWhichIsA("Tool"); if already then onEquip(already) end end
        player.CharacterAdded:Connect(function(char) activeTool = nil; task.wait(0.2); hookCharacter(char) end)
        startLoop()
    end
    function M.setEnabled(state)
        M.enabled = state and true or false
        if M.enabled then init()
            if player.Character then local tool = player.Character:FindFirstChildWhichIsA("Tool"); if tool and is_gun(tool) then onEquip(tool) end end
        else activeTool = nil end
    end
    function M.setFireDelay(v) FIRE_DELAY = math.clamp(tonumber(v) or 0.1, 0.01, 10) end
    return M
end)()

local TriggerBot = (function()
    local M = { enabled=false, initialized=false }
    local SETTINGS = { AimRadius=20, FireDelay=0.08, HPrediction=0.155, VPrediction=0.155,
        SilentAim=true, IgnoreKnocked=true, HitPart="Head", KnifeCheck=false, ForcefieldCheck=false, KnockedCheck=false }
    local last_shot = 0
    local main_remote = ReplicatedStorage:FindFirstChild("MAINEVENT") or ReplicatedStorage:FindFirstChild("MainEvent")
    local function has_char(p) return p and p.Character and p.Character:FindFirstChild("HumanoidRootPart") end
    local function is_knocked(p) if not has_char(p) then return false end; local be = p.Character:FindFirstChild("BodyEffects"); if not be then return false end; local ko = be:FindFirstChild("K.O"); return ko and ko.Value or false end
    local function local_has_katana() if not player.Character then return false end; local tool = player.Character:FindFirstChildWhichIsA("Tool"); if not tool then return false end; return string.lower(tool.Name):find("katana", 1, true) ~= nil end
    local function has_forcefield(p) if not has_char(p) then return false end; return p.Character:FindFirstChildOfClass("ForceField") ~= nil end
    local function is_knocked_hp(p) if not has_char(p) then return false end; local hum = p.Character:FindFirstChildOfClass("Humanoid"); if not hum then return false end; return hum.Health >= 1 and hum.Health <= 20 end
    local function get_equipped_gun()
        if not player.Character then return nil end
        local tool = player.Character:FindFirstChildWhichIsA("Tool")
        if not tool or not tool:FindFirstChild("Handle") then return nil end
        return tool
    end
    local function get_aim_ray()
        if UserInputService.MouseBehavior == Enum.MouseBehavior.LockCenter then return camera.CFrame.Position, camera.CFrame.LookVector end
        local mousePos = UserInputService:GetMouseLocation(); local inset = GuiService:GetGuiInset()
        local ray = camera:ViewportPointToRay(mousePos.X - inset.X, mousePos.Y - inset.Y)
        return ray.Origin, ray.Direction
    end
    local function get_aim_screen_point()
        if UserInputService.MouseBehavior == Enum.MouseBehavior.LockCenter then return Vector2.new(camera.ViewportSize.X/2, camera.ViewportSize.Y/2) end
        local mousePos = UserInputService:GetMouseLocation(); local inset = GuiService:GetGuiInset()
        return Vector2.new(mousePos.X - inset.X, mousePos.Y - inset.Y)
    end
    local function crosshair_hits_player(p)
        if not has_char(p) then return false end
        local origin, dir = get_aim_ray()
        local params = RaycastParams.new(); params.FilterType = Enum.RaycastFilterType.Exclude
        local filter = {}; if player.Character then table.insert(filter, player.Character) end
        params.FilterDescendantsInstances = filter; params.IgnoreWater = true
        local hit = workspace:Raycast(origin, dir.Unit * 1000, params)
        if not hit or not hit.Instance then return false end
        return hit.Instance:IsDescendantOf(p.Character)
    end
    local function get_aimed_player()
        local origin, dir = get_aim_ray()
        local params = RaycastParams.new(); params.FilterType = Enum.RaycastFilterType.Exclude
        local filter = {}; if player.Character then table.insert(filter, player.Character) end
        params.FilterDescendantsInstances = filter; params.IgnoreWater = true
        local hit = workspace:Raycast(origin, dir.Unit * 1000, params)
        if hit and hit.Instance then
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= player and has_char(p) and hit.Instance:IsDescendantOf(p.Character) then
                    if SETTINGS.IgnoreKnocked and is_knocked(p) then return nil end
                    return p
                end
            end
        end
        local aimScreen = get_aim_screen_point(); local best, bestDist = nil, math.min(SETTINGS.AimRadius, 22)
        for _, p in ipairs(Players:GetPlayers()) do
            if p == player then continue end; if not has_char(p) then continue end
            if SETTINGS.IgnoreKnocked and is_knocked(p) then continue end
            local part = ResolveClosestPart(p.Character, SETTINGS.HitPart)
            if part then
                local sp, on = camera:WorldToViewportPoint(part.Position)
                if on and sp.Z > 0 then
                    local d = (Vector2.new(sp.X, sp.Y) - aimScreen).Magnitude
                    if d < bestDist then bestDist, best = d, p end
                end
            end
        end
        if best and crosshair_hits_player(best) then return best end
        return nil
    end
    local function init()
        if M.initialized then return end; M.initialized = true
        RunService.Heartbeat:Connect(function()
            if not M.enabled then return end
            if tick() - last_shot < SETTINGS.FireDelay then return end
            if SETTINGS.KnifeCheck and local_has_katana() then return end
            local gun = get_equipped_gun(); if not gun then return end
            local target = get_aimed_player(); if not target then return end
            if SETTINGS.ForcefieldCheck and has_forcefield(target) then return end
            if SETTINGS.KnockedCheck and is_knocked_hp(target) then return end
            local char = target.Character; local hrp = char and char:FindFirstChild("HumanoidRootPart"); if not hrp then return end
            local aimPart = ResolveClosestPart(char, SETTINGS.HitPart) or hrp
            local pos, vel = aimPart.Position, hrp.AssemblyLinearVelocity
            local predicted = Vector3.new(pos.X + vel.X*SETTINGS.HPrediction, pos.Y + vel.Y*SETTINGS.VPrediction, pos.Z)
            last_shot = tick()
            if SETTINGS.SilentAim and main_remote then
                pcall(function() main_remote:FireServer("MOUSE", predicted) end)
                pcall(function() main_remote:FireServer("MousePos", predicted) end)
                pcall(function() main_remote:FireServer("UpdateMousePosI", predicted) end)
                pcall(function() main_remote:FireServer("UpdateMousePos", predicted) end)
            end
            pcall(function() gun:Activate() end)
        end)
    end
    function M.setEnabled(state) M.enabled = state and true or false; if M.enabled then init(); last_shot = 0 end end
    function M.setFireDelay(v) SETTINGS.FireDelay = tonumber(v) or SETTINGS.FireDelay end
    function M.setHPrediction(v) SETTINGS.HPrediction = tonumber(v) or SETTINGS.HPrediction end
    function M.setVPrediction(v) SETTINGS.VPrediction = tonumber(v) or SETTINGS.VPrediction end
    function M.setHitPart(name) SETTINGS.HitPart = name or "Head" end
    function M.setKnifeCheck(v) SETTINGS.KnifeCheck = v and true or false end
    function M.setForcefieldCheck(v) SETTINGS.ForcefieldCheck = v and true or false end
    function M.setKnockedCheck(v) SETTINGS.KnockedCheck = v and true or false end
    return M
end)()

local function FindPlayer(query)
    if not query or query == "" then return nil end
    local q = string.lower(query)
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LP then
            local name, display = string.lower(plr.Name), string.lower(plr.DisplayName or plr.Name)
            if name == q or display == q or string.find(name, q, 1, true) or string.find(display, q, 1, true) then return plr end
        end
    end
    return nil
end
TargetTab:Section("actions", 2):ButtonRow({
    { Text = "find", Callback = function()
        local plr = FindPlayer(searchState.query)
        if plr then selTarget.Player = plr; notify("target", "found " .. plr.Name, "success")
        else selTarget.Player = nil; notify("target", "not found", "error") end
    end },
    { Text = "clear", Callback = function() selTarget.Player = nil; notify("target", "cleared", "warn") end },
})

    while task.wait(0.5) do
        pcall(function()
            local p = selTarget.Player
            if p and p.Parent and p.UserId then
                if p.UserId ~= lastUserId then
                    lastUserId = p.UserId
                    loadAvatar(targetAvatar, p.UserId)
                end
                targetNameLabel:SetName("target: " .. p.Name)
                local char = p.Character
                if char then
                    local hum = char:FindFirstChildOfClass("Humanoid")
                    local be = char:FindFirstChild("BodyEffects")
                    local av = be and be:FindFirstChild("Armor")
                    local myHRP = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
                    local hrp = char:FindFirstChild("HumanoidRootPart")
                    local hp = hum and math.floor(hum.Health) or 0
                    local armor = av and math.floor(av.Value) or 0
                    local dist = (myHRP and hrp) and math.floor((myHRP.Position - hrp.Position).Magnitude) or 0
                    targetInfoLabel:SetName("health: "..hp.." | armor: "..armor.." | dist: "..dist)
                else
                    targetInfoLabel:SetName("health: — | armor: — | dist: —")
                end
            else
                if lastUserId ~= nil then
                    lastUserId = nil
                    targetAvatar.Image = "rbxasset://textures/ui/GuiImagePlaceholder.png"
                end
                targetNameLabel:SetName("no target selected")
                targetInfoLabel:SetName("health: — | armor: — | dist: —")
            end
        end)
    end
end)

local spectateActive = false
local spectateConn = nil
local function stopSpectate()
    spectateActive = false
    if spectateConn then spectateConn:Disconnect(); spectateConn = nil end
    local cam = Workspace.CurrentCamera
    local myHum = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
    if cam and myHum then cam.CameraSubject = myHum; cam.CameraType = Enum.CameraType.Custom end
end
local function startSpectate()
    if spectateConn then return end
    spectateActive = true
    spectateConn = RunService.RenderStepped:Connect(function()
        if not spectateActive then return end
        local p = selTarget.Player
        if not p or not p.Parent then stopSpectate(); return end
        local char = p.Character; if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        local subj = (hum and hum.Health > 0) and hum or char:FindFirstChild("Head")
        if subj and Workspace.CurrentCamera.CameraSubject ~= subj then Workspace.CurrentCamera.CameraSubject = subj end
    end)
end

local configs = SettingsTab:Section("configs", 1)
local configName = "default"
configs:Search({ Placeholder = "config name", Label = "name", Live = true, Callback = function(text) configName = text end })
local configDropdown = configs:Dropdown({ Text = "saved configs", Flag = "selected_config", Options = { "" }, Default = "",
    Callback = function(value) if value and value ~= "" then configName = value end end })
local function refreshConfigDropdown()
    local list = SaveManager:List(); if #list == 0 then list = { "" } end
    pcall(function() configDropdown:SetOptions(list) end)
end
refreshConfigDropdown()
configs:ButtonRow({
    { Text = "create", Callback = function()
        if configName == nil or configName == "" then notify("config", "enter a name first", "warn"); return end
        if SaveManager:Save(configName) then refreshConfigDropdown(); notify("config", "created " .. configName, "success")
        else notify("config", "failed to save " .. configName, "error") end
    end },
    { Text = "overwrite", Callback = function()
        if configName == nil or configName == "" then notify("config", "enter a name first", "warn"); return end
        if SaveManager:Save(configName) then refreshConfigDropdown(); notify("config", "overwritten " .. configName, "success")
        else notify("config", "failed to save " .. configName, "error") end
    end },
})
configs:ButtonRow({
    { Text = "load", Callback = function()
        if configName == nil or configName == "" then notify("config", "enter a name first", "warn"); return end
        if SaveManager:Load(configName) then notify("config", "loaded " .. configName, "success")
        else notify("config", "could not load " .. configName, "error") end
    end },
    { Text = "delete", Callback = function()
        if configName == nil or configName == "" then notify("config", "enter a name first", "warn"); return end
        SaveManager:Delete(configName); refreshConfigDropdown(); notify("config", "deleted " .. configName, "success")
    end },
})
configs:ButtonRow({
    { Text = "set autoload", Callback = function()
        if configName == nil or configName == "" then notify("config", "enter a name first", "warn"); return end
        SaveManager:SetAutoloadConfig(configName); refreshConfigDropdown(); notify("config", "autoload set to " .. configName, "success")

    { Text = "remove autoload", Callback = function()

    end },
})

--ui polish
task.spawn(function()
    task.wait(1)
    local function roundAll(root)
        for _, d in ipairs(root:GetDescendants()) do
            if d:IsA("Frame") or d:IsA("TextButton") or d:IsA("TextLabel") or d:IsA("ImageLabel") then
                local c = d:FindFirstChildOfClass("UICorner")
                if not c then c = Instance.new("UICorner"); c.Parent = d end
                local m = math.min(d.AbsoluteSize.X, d.AbsoluteSize.Y)
                local px = math.clamp(m * 0.12, 4, 12)
                c.CornerRadius = UDim.new(0, px)
            end
        end
    end
    local pg = LP:FindFirstChild("PlayerGui")
    if pg then
        for _, g in ipairs(pg:GetChildren()) do
            if g:IsA("ScreenGui") and #g:GetDescendants() > 100 then pcall(roundAll, g) end
        end
    end
    for _, g in ipairs(CoreGui:GetChildren()) do
        if g:IsA("ScreenGui") and #g:GetDescendants() > 100 then pcall(roundAll, g) end
    end
end)
