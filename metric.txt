local function BehindWall(player)
    if not player or player == game.Players.LocalPlayer or not player.Character then
        return true
    end

    local targetPart = player.Character:FindFirstChild("HumanoidRootPart")
    if not targetPart then
        return true
    end

    local origin = workspace.CurrentCamera.CFrame.Position
    local direction = (targetPart.Position - origin).unit * (targetPart.Position - origin).magnitude
    local raycastParams = RaycastParams.new()
    raycastParams.FilterDescendantsInstances = {game.Players.LocalPlayer.Character, player.Character}
    raycastParams.FilterType = Enum.RaycastFilterType.Blacklist

    local raycastResult = workspace:Raycast(origin, direction, raycastParams)
    return raycastResult and raycastResult.Instance ~= nil
end

local function isPlayerKO(player)
    if not player or player == game.Players.LocalPlayer or not player.Character then
        return true
    end

    local bodyEffects = player.Character:FindFirstChild("BodyEffects")
    if bodyEffects then
        local KO = bodyEffects:FindFirstChild("K.O")
        if KO and KO:IsA("BoolValue") and KO.Value then
            return true
        end
    end

    local humanoidRootPart = player.Character:FindFirstChild("HumanoidRootPart")
    if humanoidRootPart and humanoidRootPart.Anchored then
        return true
    end

    return false
end

local function findnigga()
    local fovRadius = 125
    local closestPlayer = nil
    local closestDistance = math.huge

    for _, Plr in ipairs(players:GetPlayers()) do
        if Plr == players.LocalPlayer or not Plr.Character or not Plr.Character:FindFirstChild("HumanoidRootPart") then
            continue
        end

           if metric.check.Teamcheck and isPlayerOnSameTeam(Plr) then
                continue
            end
            
            if metric.check.Friendcheck and isPlayerFriend(Plr) then
                continue
            end
            
            if metric.check.KOcheck and isPlayerKO(Plr) then
                continue
            end
            
            if metric.check.Seatedcheck and isPlayerSeated(Plr) then
                continue
            end
            
            if metric.check.Wallcheck and BehindWall(Plr) then
                continue
            end
            
            if metric.check.CamKo and isPlayerKO(Plr) then
                continue
            end

        local targetPart = Plr.Character:FindFirstChild("HumanoidRootPart")
        if targetPart then
            local screenPos, onScreen = camera:WorldToViewportPoint(targetPart.Position)
            if onScreen and (not metric.check.Wallcheck or not BehindWall(Plr)) then
                local distanceToCenter = (Vector2.new(
                    camera.ViewportSize.X / 2,
                    camera.ViewportSize.Y / 2
                ) - Vector2.new(screenPos.X, screenPos.Y)).Magnitude

                if distanceToCenter < closestDistance and distanceToCenter <= fovRadius then
                    closestPlayer = Plr
                    closestDistance = distanceToCenter
                end
            end
        end
    end

    return closestPlayer
end

end

                     velocity.X * metric.aim.prediction1,

                 )

                local shakeX = (math.random() * 2 - 1) * metric.aim.shake_x
                local shakeY = (math.random() * 2 - 1) * metric.aim.shake_y

                pred = pred + Vector3.new(shakeX, shakeY, shakeZ)

                local currentPos = camera.CFrame.Position
                local targetCFrame = CFrame.new(currentPos, pred)
                local currentDirection = camera.CFrame.LookVector
                local targetDirection = targetCFrame.LookVector
                local currentHorizontal = Vector3.new(currentDirection.X, 0, currentDirection.Z).Unit
                local targetHorizontal = Vector3.new(targetDirection.X, 0, targetDirection.Z).Unit
                local smoothX = metric.aim.smoothness1 
                local easedAlphaX = tweenservice:GetValue(
                    smoothX, 
                    metric.aim.easing_style, 
                    metric.aim.easing_dir
                )
                
                local smoothedHorizontal = currentHorizontal:Lerp(targetHorizontal, easedAlphaX)
                local currentVertical = currentDirection.Y
                local targetVertical = targetDirection.Y
                local smoothY = metric.aim.smoothness2 
                local easedAlphaY = tweenservice:GetValue(
                    smoothY, 
                    metric.aim.easing_style, 
                    metric.aim.easing_dir
                )
                
                local smoothedVertical = currentVertical + (targetVertical - currentVertical) * easedAlphaY
                local smoothedDirection = (smoothedHorizontal + Vector3.new(0, smoothedVertical, 0)).Unit
                camera.CFrame = CFrame.new(currentPos, currentPos + smoothedDirection)
            end
        end
    end
end))

runservice.Heartbeat:Connect(LPH_JIT_MAX(function()
    if not metric.aa.enabled then
        targetAirStartTime = nil
        isAutoShooting = false

        return
    end

    local target = metric.targ.target or (metric.aim and metric.aim.target)

    if not target or not target.Character then
        targetAirStartTime = nil
        isAutoShooting = false

        return
    end

    local humanoid = target.Character:FindFirstChildOfClass('Humanoid')
    local rootPart = target.Character:FindFirstChild(metric.aa.part)

    if not humanoid or not rootPart then
        targetAirStartTime = nil
        isAutoShooting = false

        return
    end
    if target.Character:FindFirstChildOfClass('ForceField') then
        return
    end

    local state = humanoid:GetState()
    local isAirborne = (state == Enum.HumanoidStateType.Freefall)

    if isAirborne then
        if not targetAirStartTime then
            targetAirStartTime = tick()
        else
            local airDuration = tick() - targetAirStartTime

            if airDuration >= metric.aa.shoot_delay and not isAutoShooting then
                isAutoShooting = true

                task.spawn(function()
                    while target and target.Character and isAutoShooting do
                        local currentHumanoid = target.Character:FindFirstChildOfClass('Humanoid')

                        if not currentHumanoid or currentHumanoid:GetState() ~= Enum.HumanoidStateType.Freefall then
                            break
                        end
                        if target.Character:FindFirstChildOfClass('ForceField') then
                            break
                        end

                        local character = localplayer.Character

                        if character then
                            local tool = character:FindFirstChildOfClass('Tool')

                            if tool and not isToolBlacklisted(tool.Name) then
                                tool:Activate()
                            elseif tool and isToolBlacklisted(tool.Name) then
                            end
                        end

                        task.wait(0.00001)
                    end

                    isAutoShooting = false
                    targetAirStartTime = nil
                end)
            end
        end
    else
        targetAirStartTime = nil
        isAutoShooting = false
    end
end))

local resolverCache = {}
local onHeartbeat = LPH_JIT_MAX(function()
    if not metric.aim.resolver then
        return
    end

    local now = tick()

    for _, plr in ipairs(players:GetPlayers())do
        if plr == localplayer then
            continue
        end
        if not (plr.Character and plr.Character:FindFirstChildOfClass('Humanoid')) then
            resolverCache[plr] = nil

            continue
        end

        local char = plr.Character
        local human = char:FindFirstChildOfClass('Humanoid')
        local watch = {
            'HumanoidRootPart',
            'Head',
            'UpperTorso',
            'LowerTorso',
            'LeftUpperArm',
            'RightUpperArm',
            'LeftLowerArm',
            'RightLowerArm',
            'LeftUpperLeg',
            'RightUpperLeg',
            'LeftLowerLeg',
            'RightLowerLeg',
            'LeftHand',
            'RightHand',
            'LeftFoot',
            'RightFoot',
        }

        for _, name in ipairs(watch)do
            local part = char:FindFirstChild(name)

            if not part then
                continue
            end

            resolverCache[plr] = resolverCache[plr] or {}

            local entry = resolverCache[plr][name]
            local curPos = part.Position

            if entry then
                local dt = now - entry.time

                if dt > 0 then
                    local rawVel = (curPos - entry.pos) / dt
                    local alpha = 0.15
                    local vel = entry.vel or rawVel

                    vel = vel:Lerp(rawVel, alpha)
                    entry.vel = vel
                    part.Velocity = vel
                    part.AssemblyLinearVelocity = vel
                end
            end

            resolverCache[plr][name] = {
                pos = curPos,
                time = now,
            }
        end
    end
end)

local TargetFuturePosition = LPH_NO_VIRTUALIZE(function()
    local target = metric.targ.target
    if not target or not target.Character then return nil end
    
    local selectedPart = metric.targ.part
    local targetPart = target.Character:FindFirstChild(selectedPart)
    if not targetPart then return nil end

    local velocity = targetPart.Velocity
    local currentPos = targetPart.Position

    local humanoid = target.Character:FindFirstChildOfClass("Humanoid")
    local jumpY = currentPos.Y
    
    if humanoid then
        local state = humanoid:GetState()
        if state == Enum.HumanoidStateType.Jumping then
            jumpY = currentPos.Y + (metric.aa.offset1)
        elseif state == Enum.HumanoidStateType.Freefall then
            jumpY = currentPos.Y + (metric.aa.offset2)
        end
    end

    local horizontalPrediction = metric.targ.prediction1
    local verticalPrediction = metric.targ.prediction2

    if metric.targ.division then
        return Vector3.new(
            currentPos.X + (velocity.X / horizontalPrediction),
            jumpY + (velocity.Y / verticalPrediction),
            currentPos.Z + (velocity.Z / horizontalPrediction)
        )
    else
        return Vector3.new(
            currentPos.X + (velocity.X * horizontalPrediction),
            jumpY + (velocity.Y * verticalPrediction),
            currentPos.Z + (velocity.Z * horizontalPrediction)
        )
    end
end)

local updatePredictionValue = LPH_NO_VIRTUALIZE(function()
    local pg = tonumber(string.split(game:GetService('Stats').Network.ServerStatsItem['Data Ping']:GetValueString(), '(')[1])
    if not pg then return end

    local function fiCl(tb)
        local clVa, clDi = nil, math.huge
        for _, en in ipairs(tb) do
            local di = math.abs(pg - en[1])
            if di < clDi then
                clDi, clVa = di, en[2]
            end
        end
        return clVa
    end

    if metric.aim.enabled and metric.aim.auto then
        local mo = metric.aim.automod or "Default"
        local predictionValue = nil

        if mo == 'Default' or mo == 'Sets' then
            predictionValue = fiCl(mo == 'Sets' and predictionTable or sigma_table)
        elseif mo == 'Math Calc' then
            predictionValue = 0.105 * math.log(pg + 1) - 0.342665
        elseif mo == 'Addition Calc' then
            predictionValue = calculate(pg)
        end
        
        if predictionValue then
            metric.aim.prediction1 = predictionValue
            metric.aim.prediction2 = predictionValue
        end
    end

    if metric.targ.enabled and metric.targ.auto then
        local mo = metric.targ.automod or "Default"
        local predictionValue = nil

        if mo == 'Default' or mo == 'Sets' then
            predictionValue = fiCl(mo == 'Sets' and predictionTable or sigma_table)
        elseif mo == 'Math Calc' then
            predictionValue = 0.105 * math.log(pg + 1) - 0.342665
        elseif mo == 'Addition Calc' then
            predictionValue = calculate(pg)
        end
        
        if predictionValue then
            metric.targ.prediction1 = predictionValue
            metric.targ.prediction2 = predictionValue
        end
    end
end)

local HitChams = LPH_NO_VIRTUALIZE(function(Player)
    if not metric.hcham.enabled then
        return
    end
    if Player and Player.Character and Player.Character:FindFirstChild('HumanoidRootPart') then
        Player.Character.Archivable = true

        local Cloned = Player.Character:Clone()
        Cloned.Name = 'Player Clone'

        local BodyParts = {
            'Head',
            'UpperTorso',
            'LowerTorso',
            'LeftUpperArm',
            'LeftLowerArm',
            'LeftHand',
            'RightUpperArm',
            'RightLowerArm',
            'RightHand',
            'LeftUpperLeg',
            'LeftLowerLeg',
            'LeftFoot',
            'RightUpperLeg',
            'RightLowerLeg',
            'RightFoot',
        }

        for _, Part in ipairs(Cloned:GetChildren())do
            if Part:IsA('BasePart') then
                local PartValid = false

                for _, validPart in ipairs(BodyParts)do
                    if Part.Name == validPart then
                        PartValid = true
                        break
                    end
                end

                if not PartValid then
                    Part:Destroy()
                end
            elseif Part:IsA('Accessory') or Part:IsA('Tool') or Part:IsA('Shirt') or Part:IsA('Pants') or Part:IsA('Hat') then
                Part:Destroy()
            end
        end

        if Cloned:FindFirstChild('Humanoid') then
            Cloned.Humanoid:Destroy()
        end

        for _, BodyPart in ipairs(Cloned:GetChildren())do
            if BodyPart:IsA('BasePart') then
                BodyPart.CanCollide = false
                BodyPart.Anchored = true
                BodyPart.Transparency = metric.hcham.transparency
                BodyPart.Color = metric.hcham.color
                BodyPart.Material = metric.hcham.material

                for _, child in ipairs(BodyPart:GetDescendants())do
                    if child:IsA('Decal') or child:IsA('Texture') or child:IsA('SurfaceAppearance') then
                        child:Destroy()
                    elseif child:IsA('SpecialMesh') then
                        child.TextureId = ''
                    elseif child:IsA('Motor6D') or child:IsA('Weld') then
                        child:Destroy()
                    end
                end

                if BodyPart:IsA('MeshPart') then
                    BodyPart.TextureID = ''
                end
            end
        end

        if Cloned:FindFirstChild('Head') then
            local Head = Cloned.Head
            Head.Transparency = metric.hcham.transparency
            Head.Color = metric.hcham.color
            Head.Material = metric.hcham.material

            for _, child in ipairs(Head:GetChildren())do
                if child:IsA('Decal') and child.Name == 'face' then
                    child:Destroy()
                end
            end
        end

        Cloned.Parent = workspace

        local tweenInfo = TweenInfo.new(metric.hcham.duration, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut, 0, false)

        for _, BodyPart in ipairs(Cloned:GetChildren())do
            if BodyPart:IsA('BasePart') then
                local tween = tweenservice:Create(BodyPart, tweenInfo, {Transparency = 1})

                tween:Play()
            end
        end

        task.delay(metric.hcham.duration + 0.1, function()
            if Cloned and Cloned.Parent then
                Cloned:Destroy()
            end
        end)
    end
end)

        last_health[target.UserId] = current_health
    end

    if metric.hcham.enabled and target and humanoid then
        local current_health = humanoid.Health
        last_health_chams[target.UserId] = last_health_chams[target.UserId] or current_health

        if current_health < last_health_chams[target.UserId] then
            HitChams(target)
        end

        last_health_chams[target.UserId] = current_health
    else
        for userId in pairs(last_health_chams) do
            if not game.Players:GetPlayerByUserId(userId) then
                last_health_chams[userId] = nil
            end
        end
    end        
end))


local NEWFrame = LPH_NO_VIRTUALIZE(function()
    local target = metric.targ and metric.targ.target

    if not target or not target.Character or not target.Character:FindFirstChild('HumanoidRootPart') then
        return nil
    end

    local targetPosition = target.Character.HumanoidRootPart.Position

    if metric.csync.Type == 'Random' then
        return CFrame.new(targetPosition + Vector3.new(
            math.random(-metric.csync.RandomAmount, metric.csync.RandomAmount), 
            math.random(0, metric.csync.RandomAmount), 
            math.random(-metric.csync.RandomAmount, metric.csync.RandomAmount)
        )) * CFrame.Angles(
            math.rad(math.random(0, 360)), 
            math.rad(math.random(0, 360)), 
            math.rad(math.random(0, 360))
        )
        
    elseif metric.csync.Type == 'Orbit' then
        local CurrentTime = tick()
        return CFrame.new(targetPosition) * 
               CFrame.Angles(0, 2 * math.pi * CurrentTime * metric.csync.Speed % (2 * math.pi), 0) * 
               CFrame.new(0, metric.csync.Height, metric.csync.Distance)
               
    elseif metric.csync.Type == 'Spiral' then
        local CurrentTime = tick()
        local angle = 2 * math.pi * CurrentTime * metric.csync.Speed % (2 * math.pi)
        local spiralRadius = metric.csync.Distance + math.sin(CurrentTime) * metric.csync.RandomAmount

        return CFrame.new(targetPosition) * CFrame.Angles(0, angle, 0) * 
               CFrame.new(spiralRadius, metric.csync.Height, 0)
               
    elseif metric.csync.Type == 'Spherical' then
        local CurrentTime = tick()
        local radius = metric.csync.Distance
        local sphericalX = radius * math.sin(CurrentTime) * math.cos(CurrentTime * metric.csync.Speed)
        local sphericalY = radius * math.sin(CurrentTime) * math.sin(CurrentTime * metric.csync.Speed)
        local sphericalZ = radius * math.cos(CurrentTime)

        return CFrame.new(targetPosition + Vector3.new(sphericalX, sphericalY, sphericalZ))
        
    elseif metric.csync.Type == 'Attach' then
        if target and target.Character and target.Character:FindFirstChild(metric.targ.part) then
            local part = target.Character[metric.targ.part]
            local predicted = TargetFuturePosition()

            if metric.aa and metric.aa.enabled then
                predicted = predicted + Vector3.new(0, metric.aa.offset1 + metric.aa.offset2, 0)
            end

            return CFrame.new(predicted)
        end
    end

    return nil
end)

local function solve(angle, radius)
    return Vector2.new(math.sin(math.rad(angle)) * radius, math.cos(math.rad(angle)) * radius)
end

                    )

                text_2.Color = crosshair.color

                    local inline = drawings.crosshair[idx + 4]

                    local angle = (idx - 1) * 90
                    local length = crosshair.length

                    if crosshair.spin then
                        local spin_angle = -_tick * crosshair.spin_speed % crosshair.spin_max
                        angle =
                            angle +
                            tweenservice:GetValue(spin_angle / 360, crosshair.spin_style, Enum.EasingDirection.InOut) *
                                360
                    end

                    if crosshair.resize then
                        local resize_length = tick() * crosshair.resize_speed % 180
                        length = crosshair.resize_min + math.sin(math.rad(resize_length)) * crosshair.resize_max
                    end

                    inline.Visible = true
                    inline.Color = crosshair.color
                    inline.From = position + solve(angle, crosshair.radius)
                    inline.To = position + solve(angle, crosshair.radius + length)
                    inline.Thickness = crosshair.width

                    outline.Visible = true
                    outline.From = position + solve(angle, crosshair.radius - 1)
                    outline.To = position + solve(angle, crosshair.radius + length + 1)
                    outline.Thickness = crosshair.width + 1.5
                end

                    drawings.crosshair[idx + 4].Visible = false
                end

local function findAmmoValue(tool)
    if not tool then return nil, nil end
    
    for _, child in ipairs(tool:GetChildren()) do
        local lowerName = child.Name:lower()
        if lowerName:find("ammo") then
            if child:IsA("IntValue") or child:IsA("NumberValue") then
                return child, child.Value
            end
            for _, subChild in ipairs(child:GetChildren()) do
                if subChild:IsA("IntValue") or subChild:IsA("NumberValue") then
                    return subChild, subChild.Value
                end
            end
        end
    end

    local values = tool:FindFirstChild("Values")
    if values then
        for _, child in ipairs(values:GetChildren()) do
            local lowerName = child.Name:lower()
            if lowerName:find("ammo") and (child:IsA("IntValue") or child:IsA("NumberValue")) then
                return child, child.Value
            end
        end
    end

    for attrName, attrValue in pairs(tool:GetAttributes()) do
        local lowerName = attrName:lower()
        if lowerName:find("ammo") and type(attrValue) == "number" then
            return attrName, attrValue
        end
    end

    for _, child in ipairs(tool:GetChildren()) do
        if child.Name:lower():find("stats") or child.Name:lower():find("data") then
            for _, statChild in ipairs(child:GetChildren()) do
                local lowerName = statChild.Name:lower()
                if lowerName:find("ammo") and (statChild:IsA("IntValue") or statChild:IsA("NumberValue")) then
                    return statChild, statChild.Value
                end
            end
        end
    end
    
    return nil, nil
end

local function toolUsesAmmo(tool)
    if not tool then return false end

    for _, child in ipairs(tool:GetDescendants()) do
        if child.Name:lower():find("ammo") then
            return true
        end
    end

    for attrName, _ in pairs(tool:GetAttributes()) do
        if attrName:lower():find("ammo") then
            return true
        end
    end
    
    return false
end

local function pressReloadKey()
    VIM:SendKeyEvent(true, Enum.KeyCode.R, false, game)
    task.wait(0.05)
    VIM:SendKeyEvent(false, Enum.KeyCode.R, false, game)
    
    if metric.autoreload.debug then
        warn("[AutoReload] Pressed R key")
    end
end

local function fireReload(tool)
    local currentTime = tick()
    if currentTime - metric.autoreload.last_reload < metric.autoreload.reload_cooldown then
        return
    end
    
    metric.autoreload.last_reload = currentTime
    
    if metric.autoreload.debug then
        warn("[AutoReload] Attempting reload for: " .. (tool and tool.Name or "Unknown"))
    end
    
    local reloaded = false
    
    if MainEvent then
        local reloadArgs = {
            "Reload",
            "ReloadWeapon", 
            "ReloadGun",
            "RELOAD",
            "ReloadRequest",
            "r",
            "Rel",
            "ReloadAmmo",
            "reload",
            "Reloading"
        }
        
        for _, arg in ipairs(reloadArgs) do
            local success = pcall(function()
                MainEvent:FireServer(arg)
            end)
            if success then
                if metric.autoreload.debug then
                    warn("[AutoReload] Success with arg: " .. arg)
                end
                reloaded = true
                break
            end
        end
        
        if not reloaded and tool then
            local success = pcall(function()
                MainEvent:FireServer("Reload", tool.Name)
            end)
            if success then
                reloaded = true
                if metric.autoreload.debug then warn("[AutoReload] Success with tool name") end
            end
        end
    end
    
    if not reloaded then
        if metric.autoreload.debug then
            warn("[AutoReload] Remote methods failed, falling back to R key")
        end
        pressReloadKey()
    end
    
    if metric.targ.logs then
        Library:Notication("Auto Reloaded", 0.5)
    end
    
    return true
end

local lastTool = nil
local ammoConnection = nil
local reloadRunning = false

local function cleanupReloadDetection()
    if ammoConnection then
        ammoConnection:Disconnect()
        ammoConnection = nil
    end
    lastTool = nil
    reloadRunning = false
end

local function setupReloadTool(tool)
    if not tool then return end
    if not toolUsesAmmo(tool) then 
        if metric.autoreload.debug then
            warn("[AutoReload] No ammo detected in: " .. tool.Name)
        end
        return 
    end
    
    local ammoObj, currentAmmo = findAmmoValue(tool)
    if not ammoObj then 
        if metric.autoreload.debug then
            warn("[AutoReload] Found ammo reference but couldn't read value in: " .. tool.Name)
        end
        return 
    end
    
    if metric.autoreload.debug then
        warn("[AutoReload] Setup for: " .. tool.Name .. " | Ammo: " .. tostring(currentAmmo))
    end
    
    if type(currentAmmo) == "number" and currentAmmo <= 0 then
        task.wait(metric.autoreload.reload_delay)
        fireReload(tool)
        return
    end

    if typeof(ammoObj) == "Instance" then
        ammoConnection = ammoObj:GetPropertyChangedSignal("Value"):Connect(function()
            if not metric.autoreload.enabled then return end
            
            local newValue = ammoObj.Value
            if type(newValue) == "number" and newValue <= 0 then
                task.wait(metric.autoreload.reload_delay)
                fireReload(tool)
            end
        end)
    end
    
    reloadRunning = true
    task.spawn(function()
        while metric.autoreload.enabled and tool.Parent == localplayer.Character and reloadRunning do
            local _, ammo = findAmmoValue(tool)
            
            if type(ammo) == "number" and ammo <= 0 then
                fireReload(tool)
                task.wait(0.5)
            else
                task.wait(metric.autoreload.check_interval)
            end
        end
    end)
end

local function onCharReload(char)
    char.ChildAdded:Connect(function(child)
        if child:IsA("Tool") then
            task.wait(0.1)
            
            if not metric.autoreload.enabled then return end
            
            cleanupReloadDetection()
            lastTool = child
            setupReloadTool(child)
        end
    end)

    char.ChildRemoved:Connect(function(child)
        if child:IsA("Tool") and child == lastTool then
            cleanupReloadDetection()
        end
    end)

    task.spawn(function()
        task.wait(0.1)
        if not metric.autoreload.enabled then return end
        
        local currentTool = char:FindFirstChildOfClass("Tool")
        if currentTool and toolUsesAmmo(currentTool) then
            lastTool = currentTool
            setupReloadTool(currentTool)
        end
    end)
end

local function AutoStomp()
    if not metric.autostomp.enabled then
        return
    end

    local target = metric.autostomp.target
    if metric.autostomp.require_aimlock then
        target = metric.targ.target or metric.aim.target
    end
    
    if not target or not target.Character then
        return
    end

    local bodyEffects = target.Character:FindFirstChild("BodyEffects")
    if not bodyEffects then
        return
    end

    local knockOut = bodyEffects:FindFirstChild("K.O")
    local isDead = bodyEffects:FindFirstChild("SDeath")

    if not (knockOut and knockOut.Value == true) then
        return
    end
    
    if isDead and isDead.Value == true then
        return
    end

    local currentTime = tick()
    if currentTime - metric.autostomp.last_stomp < metric.autostomp.cooldown then
        return
    end

    local character = localplayer.Character
    if not character then
        return
    end

    local humanoidRootPart = character:FindFirstChild("HumanoidRootPart")
    if not humanoidRootPart then
        return
    end

    local savedPosition = humanoidRootPart.CFrame

    local upperTorso = target.Character:FindFirstChild("UpperTorso")
    if not upperTorso then
        return
    end

    humanoidRootPart.CFrame = CFrame.new(upperTorso.Position + Vector3.new(0, 3, 0))

    if MainEvent then
        metric.autostomp.last_stomp = currentTime
        
        local stompArgs = {"Stomp", "STOMP", "stomp", "StompPlayer", "StompRequest", "s", "S", "stmp", "STMP", "stm", "STM"}
        
        for _, arg in ipairs(stompArgs) do
            local success = pcall(function()
                MainEvent:FireServer(arg)
            end)
            if success then break end
        end

        if metric.targ.logs and Library then
            Library:Notification('Trying To Stomp <font color="'..RealColor..'">'..target.DisplayName..'</font>', metric.targ.notify_duration)
        end
    end

    task.delay(metric.autostomp.stomp_delay, function()
        humanoidRootPart.CFrame = savedPosition
    end)
end

runservice.Heartbeat:Connect(LPH_JIT_MAX(function()
    if metric.autostomp.enabled then
        AutoStomp()
    end
end))

local cframe_to_offset = function(origin, target)
    local actual_origin = origin * CFrame.new(0, -1, 0, 1, 0, 0, 0, 0, 1, 0, -1, 0)

    return actual_origin:ToObjectSpace(target):inverse()
end

bool_at_tp_tool = function(tool)
    local old_grip = tool.Grip
    local target = metric.targ and metric.targ.target

    if target and target.Character and target.Character:FindFirstChild('HumanoidRootPart') then
        local character = localplayer.Character

        if not character or not character:FindFirstChild('RightHand') then
            return
        end

        local rightHand = character.RightHand

        tool.Parent = localplayer.Backpack
        rightHand.Anchored = false
        tool.Grip = cframe_to_offset(rightHand.CFrame, target.Character.HumanoidRootPart.CFrame)
        rightHand.Anchored = true
        tool.Parent = character

        runservice.RenderStepped:Wait()

        tool.Parent = localplayer.Backpack
        rightHand.Anchored = false
        tool.Grip = old_grip
        tool.Parent = character
    end
end

local function createCornerBoxLines()
    local lines = {}
    for i = 1, 8 do
        local line = Drawing.new("Line")
        line.Thickness = 3
        line.Visible = false
        lines[i] = line
    end
    return lines
end

local function drawCornerBox(lines, x, y, w, h, outlineColor, filled, fillColor)
    local c = math.min(w, h) * 0.25
    
    lines[1].From = Vector2.new(x, y); lines[1].To = Vector2.new(x + c, y)
    lines[2].From = Vector2.new(x, y); lines[2].To = Vector2.new(x, y + c)
    lines[3].From = Vector2.new(x + w - c, y); lines[3].To = Vector2.new(x + w, y)
    lines[4].From = Vector2.new(x + w, y); lines[4].To = Vector2.new(x + w, y + c)
    lines[5].From = Vector2.new(x, y + h - c); lines[5].To = Vector2.new(x, y + h)
    lines[6].From = Vector2.new(x, y + h); lines[6].To = Vector2.new(x + c, y + h)
    lines[7].From = Vector2.new(x + w - c, y + h); lines[7].To = Vector2.new(x + w, y + h)
    lines[8].From = Vector2.new(x + w, y + h - c); lines[8].To = Vector2.new(x + w, y + h)
    
    for i = 1, 8 do 
        lines[i].Color = outlineColor
        lines[i].Visible = true
        lines[i].Thickness = 2
        lines[i].Transparency = 1
    end

    if filled then
        if not lines.backgroundFill then
            lines.backgroundFill = Drawing.new("Square")
        end
        
        local fillRect = lines.backgroundFill
        fillRect.Size = Vector2.new(w, h)
        fillRect.Position = Vector2.new(x, y)
        fillRect.Color = fillColor
        fillRect.Transparency = 0.3
        fillRect.Filled = true
        fillRect.Visible = true
        fillRect.Thickness = 0 
    else
        if lines.backgroundFill then
            lines.backgroundFill.Visible = false
        end
    end
end

local function updatePlayerCache(player)
    local cache = playerCache[player]
    if not cache then return false end
    
    local now = tick()
    if now - cache.lastUpdate < G.ESPsc.RefreshRate then
        return cache.onScreen
    end
    cache.lastUpdate = now
    
    cache.character = player.Character
    if not cache.character then
        cache.onScreen = false
        return false
    end
    
    cache.rootPart = cache.character:FindFirstChild("HumanoidRootPart")
    cache.humanoid = cache.character:FindFirstChildOfClass("Humanoid")
    
    if not cache.rootPart or not cache.humanoid then
        cache.onScreen = false
        return false
    end
    
    cache.screenPos, cache.onScreen = camera:WorldToViewportPoint(cache.rootPart.Position)
    if not cache.onScreen then return false end
    
    cache.distance = (cache.rootPart.Position - camera.CFrame.Position).Magnitude
    local tool = cache.character:FindFirstChildOfClass("Tool")
    cache.toolName = tool and tool.Name or "None"
    
    return true
end

local function lerpColor(a, b, t)
    return Color3.new(
        a.R + (b.R - a.R) * t,
        a.G + (b.G - a.G) * t,
        a.B + (b.B - a.B) * t
    )
end

local function resetBoxVisuals(d)
    if not d then return end

    if d.box then
        d.box.Visible = false
    end

    if d.boxGradient then
        for _, layer in ipairs(d.boxGradient) do
            layer.Visible = false
        end
    end

    if d.cornerLines then
        for _, line in ipairs(d.cornerLines) do
            if line then
                line.Visible = false
            end
        end

        if d.cornerLines.backgroundFill then
            d.cornerLines.backgroundFill.Visible = false
        end
    end

    if d.cornerGradient then
        for _, group in ipairs(d.cornerGradient) do
            for _, sq in ipairs(group) do
                sq.Visible = false
            end
        end
    end
end

                local ping = math.abs(math.sin((G.ESPsc.Box._phase or 0) * math.pi))
                c1 = lerpColor(G.ESPsc.Box.FilledColor, G.ESPsc.Box.GradientFill, ping)
                c2 = lerpColor(G.ESPsc.Box.GradientFill, G.ESPsc.Box.FilledColor, ping)
            else
                c1 = G.ESPsc.Box.FilledColor
                c2 = G.ESPsc.Box.GradientFill
            end

            for i, layer in ipairs(d.boxGradient) do
                local t      = (i / #d.boxGradient) ^ 1.25
                local offset = i - 1
                local w = boxWidth  - offset * 2
                local h = boxHeight - offset * 2

                if w <= 0 or h <= 0 then
                    layer.Visible = false
                    continue
                end

                layer.Size     = Vector2.new(w, h)
                layer.Position = Vector2.new(
                    boxTopLeft.X + offset,
                    boxTopLeft.Y + offset
                )
                layer.Color        = lerpColor(c1, c2, t)
                layer.Transparency = G.ESPsc.Box.FillTransparency or 0.18
                layer.Visible      = true
            end
        else
            for _, g in ipairs(d.boxGradient) do g.Visible = false end
        end

        if d.cornerLines then        
            for _, line in ipairs(d.cornerLines) do
                if line then
                    line.Visible = false
                end
            end

            if d.cornerLines.backgroundFill then
                d.cornerLines.backgroundFill.Visible = false
            end
        end
    end

    if G.ESPsc.Box.Enabled and G.ESPsc.Box.Mode == "Corners" then
        d.boxOutline.Visible = false

        drawCornerBox(
            d.cornerLines,
            boxTopLeft.X,
            boxTopLeft.Y,
            boxWidth,
            boxHeight,
            G.ESPsc.Box.Color,
            G.ESPsc.Box.Filled,
            G.ESPsc.Box.FilledColor
        )

        if G.ESPsc.Box.Filled then
            for i, layer in ipairs(d.boxGradient) do
                local t = (i / #d.boxGradient) ^ 1.25
                local offset = i - 1

                local w = boxWidth - offset * 2
                local h = boxHeight - offset * 2
                if w <= 0 or h <= 0 then
                    layer.Visible = false
                else
                    layer.Size = Vector2.new(w, h)
                    layer.Position = Vector2.new(
                        boxTopLeft.X + offset,
                        boxTopLeft.Y + offset
                    )

                    layer.Color = lerpColor(
                        G.ESPsc.Box.FilledColor,
                        G.ESPsc.Box.GradientFill,
                        t
                    )

                    layer.Transparency = math.clamp(cache.distance / 2000, 0.08, 0.25)
                    layer.Visible = true
                end
             end

            for _, group in ipairs(d.cornerGradient) do
                for _, sq in ipairs(group) do
                    sq.Visible = false
                end
            end
        else
            for _, g in ipairs(d.boxGradient) do g.Visible = false end
            for _, group in ipairs(d.cornerGradient) do
                 for _, sq in ipairs(group) do sq.Visible = false end
            end
            if d.cornerLines.backgroundFill then d.cornerLines.backgroundFill.Visible = false end
        end
    end
    
    if not G.ESPsc.Box.Enabled then
        d.boxOutline.Visible = false
        if d.boxFill then d.boxFill.Visible = false end
        for _, g in ipairs(d.boxGradient) do
            g.Visible = false
        end
        for i = 1, 8 do d.cornerLines[i].Visible = false end
    end

    if G.ESPsc.Healthbar.Enabled and isTarget then
        d.healthBarBG.Visible = true
        d.healthBar.Visible = true
        
        local realHP = math.clamp(cache.humanoid.Health / cache.humanoid.MaxHealth, 0, 1)
        cache.smoothHP = cache.smoothHP + (realHP - cache.smoothHP) * cache.drainSpd
        if math.abs(cache.smoothHP - realHP) < 0.001 then cache.smoothHP = realHP end
        
        local barWidth = 5
        local outlineW = 7
        local outlineH = boxHeight + 1
        local x = boxTopLeft.X - 7
        local y = boxTopLeft.Y - 1
        
        d.healthBarBG.Size = Vector2.new(outlineW, outlineH)
        d.healthBarBG.Position = Vector2.new(x, y)
        d.healthBarBG.Color = Color3.fromRGB(0, 0, 0)
        
        d.healthBar.Size = Vector2.new(barWidth, (boxHeight - 2) * cache.smoothHP)
        d.healthBar.Position = Vector2.new(x + 1, y + outlineH - ((boxHeight - 2) * cache.smoothHP))
        
        if G.ESPsc.Healthbar.DynamicColor then
            d.healthBar.Color = cache.smoothHP >= 0.7 and Color3.fromRGB(0,210,0) or
                              cache.smoothHP >= 0.4 and Color3.fromRGB(255,165,0) or
                              Color3.fromRGB(255,0,0)
        else
            d.healthBar.Color = getgenv().ESPsc.Healthbar.Color
        end
        
    else
        d.healthBarBG.Visible = false; d.healthBar.Visible = false;
    end

    if G.ESPsc.Name.Enabled and isTarget then    
        d.nameText.Visible = true
        local displayName = player.DisplayName or player.Name
        local realName = player.Name
        d.nameText.Text = string.format("%s (@%s)", displayName, realName)
        d.nameText.Position = Vector2.new(boxCenterX, boxTopLeft.Y - 16)
        d.nameText.Color = G.ESPsc.Name.Color
        d.nameText.Size = G.ESPsc.Name.Size
    else
        d.nameText.Visible = false
    end

    if G.ESPsc.Tools.Enabled and isTarget then
        d.toolText.Visible = true
        d.toolText.Text = cache.toolName
        d.toolText.Position = Vector2.new(boxCenterX, boxTopLeft.Y + boxHeight + 2)
        d.toolText.Color = getgenv().ESPsc.Tools.Color
    else
        d.toolText.Visible = false
    end

    if G.ESPsc.Distance.Enabled and isTarget then
        d.distanceText.Visible = true
        d.distanceText.Text = string.format("%dm", cache.distance)
        d.distanceText.Position = Vector2.new(boxCenterX, boxTopLeft.Y + boxHeight + 17)
        d.distanceText.Color = G.ESPsc.Distance.Color
    else
        d.distanceText.Visible = false
    end

    if G.ESPsc.Skeleton.Enabled and isTarget then
        for i, bone in ipairs(R15_BONES) do
            local partA = cache.character:FindFirstChild(bone[1])
            local partB = cache.character:FindFirstChild(bone[2])
            local line = d.skeleton[i]
            
            if partA and partB and line then
                local posA, onA = camera:WorldToViewportPoint(partA.Position)
                local posB, onB = camera:WorldToViewportPoint(partB.Position)
                
                if onA and onB then
                    line.From = Vector2.new(posA.X, posA.Y)
                    line.To = Vector2.new(posB.X, posB.Y)
                    line.Color = getgenv().ESPsc.Skeleton.Color
                    line.Visible = true
                else
                    line.Visible = false
                end
            elseif line then
                line.Visible = false
            end
        end
    else
        for _, line in ipairs(d.skeleton) do line.Visible = false end
    end

    if G.ESPsc.Tracer.Enabled and isTarget then
        d.tracer.Visible = true
        local viewport = camera.ViewportSize
        
        local fromPos = (G.ESPsc.Tracer.Mode == "Top" and Vector2.new(viewport.X / 2, 0)) or
                       (G.ESPsc.Tracer.Mode == "Center" and Vector2.new(viewport.X / 2, viewport.Y / 2)) or
                       (G.ESPsc.Tracer.Mode == "Bottom" and Vector2.new(viewport.X / 2, viewport.Y))
        
        d.tracer.From = fromPos
        d.tracer.To = Vector2.new(cache.screenPos.X, cache.screenPos.Y)
        d.tracer.Color = G.ESPsc.Tracer.Color
        d.tracer.Thickness = G.ESPsc.Tracer.Thickness
    else
        d.tracer.Visible = false
    end

    local highlightEnabled = getgenv().ESPsc.Highlight.Enabled and isTarget
    cache.highlight.Enabled = highlightEnabled
    
    if highlightEnabled then
        cache.highlight.Adornee = cache.character
        cache.highlight.FillColor = getgenv().ESPsc.Highlight.FillColor
        cache.highlight.OutlineColor = getgenv().ESPsc.Highlight.OutlineColor
        cache.highlight.FillTransparency = getgenv().ESPsc.Highlight.FillTransparency
        cache.highlight.OutlineTransparency = getgenv().ESPsc.Highlight.OutlineTransparency
    else
        cache.highlight.Adornee = nil
    end
end)

local function hideNormalBox(d)
    if d.boxOutline then
        d.boxOutline.Visible = false
    end

    if d.boxFill then
        d.boxFill.Visible = false
    end

    if d.boxGradient then
        for _, layer in ipairs(d.boxGradient) do
            layer.Visible = false
        end
    end
end

local function hideCornerFill(d)
    if not d then return end

    if d.cornerGradient then
        for _, group in ipairs(d.cornerGradient) do
            for _, sq in ipairs(group) do
                sq.Visible = false
            end
        end
    end

    if d.cornerLines and d.cornerLines.backgroundFill then
        d.cornerLines.backgroundFill.Visible = false
    end

    if d.boxGradient then
        for _, layer in ipairs(d.boxGradient) do
            layer.Visible = false
        end
    end
end

local function clearPlayerCache(player)
    local cache = playerCache[player]
    if not cache then return end
    
    local d = cache.drawings
    d.box:Remove()
    d.boxOutline:Remove()
    d.boxFill:Remove()
    hideNormalBox(d)
    resetBoxVisuals(d)
    hideCornerFill(d)
    d.healthBarBG:Remove(); d.healthBar:Remove()
    d.nameText:Remove(); d.toolText:Remove()
    d.distanceText:Remove();
    d.tracer:Remove()
    for _, line in ipairs(d.skeleton) do line:Remove() end
    
    cache.highlight:Destroy()
    playerCache[player] = nil
end

    local bodyeffects = character:WaitForChild("BodyEffects", 1)

do
    local RunService = game:GetService("RunService")
    local Players = game:GetService("Players")
    
    local player = Players.LocalPlayer
    local character = player.Character or player.CharacterAdded:Wait()
    local humanoid = character:WaitForChild("Humanoid")
    local rootPart = character:WaitForChild("HumanoidRootPart")
    
    local lastPosition = rootPart.Position
    local lastMoveTime = tick()
    local isAFK = false
    local lastAFKState = false
    local afkConnection = nil
    
    local checkAFK = LPH_JIT_MAX(function()
        if not character or not rootPart or not humanoid then return end
        
        local currentPos = rootPart.Position
        local distanceMoved = (currentPos - lastPosition).Magnitude
        
        if distanceMoved > 0.1 then
            lastMoveTime = tick()
            lastPosition = currentPos
            isAFK = false
        else
            if tick() - lastMoveTime >= 540 then
                isAFK = true
            end
        end
    end)
    
    local moveCharacter = LPH_JIT_MAX(function()
        if not character or not rootPart or not humanoid then return end
        
        local moveDirection = Vector3.new(
            math.random(-5, 5),
            0,
            math.random(-5, 5)
        )
        
        local originalCFrame = rootPart.CFrame
        local targetCFrame = originalCFrame + moveDirection
        
        rootPart.CFrame = targetCFrame
        
        task.wait(0.05)
        
        rootPart.CFrame = originalCFrame
        
        lastMoveTime = tick()
        lastPosition = rootPart.Position
        isAFK = false
    end)
    
    local startAntiAFK = LPH_JIT_MAX(function()
        if afkConnection then return end
        
        afkConnection = RunService.Heartbeat:Connect(function()
            if not metric or not metric.misc or not metric.misc.antiafk then return end
            
            checkAFK()
            
            if isAFK then
                moveCharacter()
            end
        end)
    end)
    
    local stopAntiAFK = LPH_JIT_MAX(function()
        if afkConnection then
            afkConnection:Disconnect()
            afkConnection = nil
        end
        lastMoveTime = tick()
        isAFK = false
    end)
    
    RunService.Heartbeat:Connect(LPH_JIT(function()
        local currentState = metric and metric.misc and metric.misc.antiafk or false
        
        if currentState ~= lastAFKState then
            lastAFKState = currentState
            if currentState then
                startAntiAFK()
            else
                stopAntiAFK()
            end
        end
    end))
    
    player.CharacterAdded:Connect(function(newChar)
        character = newChar
        humanoid = newChar:WaitForChild("Humanoid")
        rootPart = newChar:WaitForChild("HumanoidRootPart")
        lastPosition = rootPart.Position
        lastMoveTime = tick()
        isAFK = false
        
        if metric and metric.misc and metric.misc.antiafk then
            startAntiAFK()
        end
    end)
end

local function UpdateOutfitList()

    local Outfits = {}
    local userId = Players.LocalPlayer.UserId
    local url = "https://avatar.roblox.com/v1/users/" .. userId .. "/outfits"

    local success, response = pcall(function()
        return HttpService:JSONDecode(game:HttpGet(url))
    end)

    if success and response and response.data then
        for _, outfit in ipairs(response.data) do
            if outfit.isEditable then
                table.insert(Outfits, outfit.name)
                CurrentOutfits[outfit.name] = outfit.id
            end
        end

        if #Outfits ~= #OutfitList.Options or table.concat(Outfits) ~= table.concat(OutfitList.Options) then
            OutfitList.Options = Outfits
            if OutfitList.Refresh then
                OutfitList:Refresh(Outfits)
            end
        end
    end
end

        if result == Enum.AvatarPromptResult.Success then
            Players.LocalPlayer.Character.Humanoid.Health = 0
        end
    end
end
